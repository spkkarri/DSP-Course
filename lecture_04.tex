\section{Eigenfunctions \& The Frequency Response}
Complex exponentials $e^{j\omega n}$ are eigenfunctions of linear time-invariant systems.
Let the input be $x[n] = e^{j\omega n}$ for $-\infty < n < \infty$. The output $y[n]$ is:
\begin{equation}
y[n] = \sum_{k=-\infty}^{\infty} h[k] x[n-k] = \sum_{k=-\infty}^{\infty} h[k] e^{j\omega(n-k)} = e^{j\omega n} \sum_{k=-\infty}^{\infty} h[k] e^{-j\omega k}
\end{equation}
We define the complex scaling factor (eigenvalue) as the frequency response $H(e^{j\omega})$:
\begin{equation}
y[n] = H(e^{j\omega}) e^{j\omega n} \quad \text{where} \quad H(e^{j\omega}) = \sum_{k=-\infty}^{\infty} h[k] e^{-j\omega k}
\end{equation}

\section{Magnitude \& Phase Responses}
The frequency response can be written in polar form:
\begin{equation}
H(e^{j\omega}) = |H(e^{j\omega})| e^{j\theta(\omega)}
\end{equation}
where $|H(e^{j\omega})|$ is the magnitude response and $\theta(\omega) = \angle H(e^{j\omega})$ is the phase response (Figure~\ref{fig:mag_phase_lpf_hpf}).
For a sinusoidal input $x[n] = A \cos(\omega_0 n + \phi)$, the output is:
\begin{equation}
y[n] = A |H(e^{j\omega_0})| \cos(\omega_0 n + \phi + \theta(\omega_0))
\end{equation}

\section{Phase Delay vs. Group Delay}
\subsection{Mathematical Definitions \& Intuitive Meaning}

\begin{table}[H]
\centering
\small
\begin{tabular}{@{}p{0.20\textwidth}p{0.32\textwidth}p{0.22\textwidth}p{0.20\textwidth}@{}}
\toprule
\textbf{Concept} & \textbf{Simplest Intuitive Meaning} & \textbf{Formula} & \textbf{Everyday Analogy} \\ \midrule
\textbf{Phase Delay ($\tau_p$)} & Time delay of a \textbf{SINGLE pure sine wave} & $\tau_p(\omega) = -\frac{\theta(\omega)}{\omega}$ & Delay of a runner's stride rhythm \\
\textbf{Group Delay ($\tau_g$)} & Time delay of the \textbf{OVERALL ENVELOPE / PACKET} & $\tau_g(\omega) = -\frac{d\theta(\omega)}{d\omega}$ & Delay of the entire group / pack of runners \\ \bottomrule
\end{tabular}
\caption{Intuitive distinction between Phase Delay and Group Delay.}
\end{table}

\subsubsection{\texorpdfstring{Intuition Behind Phase Delay ($\tau_p$)}{Intuition Behind Phase Delay}}
\begin{itemize}
    \item \textbf{Core Question:} \textit{"How long does one individual frequency component take to travel through the filter?"}
    \item \textbf{Derivation:} Sinusoidal phase shift is $\theta(\omega) = -\omega \cdot \tau_p \implies \tau_p(\omega) = -\frac{\theta(\omega)}{\omega}$ (in samples or seconds).
\end{itemize}

\subsubsection{\texorpdfstring{Intuition Behind Group Delay ($\tau_g$)}{Intuition Behind Group Delay}}
\begin{itemize}
    \item \textbf{Core Question:} \textit{"How long does the overall shape, modulation burst, or pulse envelope take to pass through the filter?"}
    \item \textbf{Why Derivative $\frac{d\theta}{d\omega}$?} Two nearby frequencies ($\omega_1, \omega_2$) interact to form a beat envelope. The delay of this envelope is $\text{Delay} \approx -\frac{\Delta \theta}{\Delta \omega}$. Taking $\Delta \omega \to 0$ yields the exact derivative $\tau_g(\omega) = -\frac{d\theta(\omega)}{d\omega}$.
\end{itemize}

\subsection{Narrowband Envelope Propagation}
Consider a narrowband wave packet input:
\begin{equation}
x[n] = s[n] \cos(\omega_0 n)
\end{equation}
Expanding the phase response $\theta(\omega)$ about $\omega_0$:
\begin{equation}
\theta(\omega) \approx \theta(\omega_0) - \tau_g(\omega_0)(\omega - \omega_0) = -\omega_0 \tau_p(\omega_0) - \tau_g(\omega_0)(\omega - \omega_0)
\end{equation}
The resulting system output is:
\begin{equation}
y[n] \approx |H(e^{j\omega_0})| s[n - \tau_g(\omega_0)] \cos(\omega_0(n - \tau_p(\omega_0)))
\end{equation}
Thus, the modulation envelope $s[n]$ is delayed by the group delay $\tau_g(\omega_0)$, while the carrier fine structure is delayed by the phase delay $\tau_p(\omega_0)$ (Figure~\ref{fig:carrier_envelope_delay}).

\section{Impact of Phase \& Group Delay in Time and Frequency Domains}
\begin{itemize}
    \item \textbf{Single Pure Sinusoid ($\omega_0$):} In the time domain, the waveform experiences a uniform time shift of $\tau_p(\omega_0)$ samples without shape distortion. In the frequency domain, the spectral line acquires a phase shift $\theta(\omega_0) = -\omega_0 \tau_p(\omega_0)$.
    \item \textbf{Narrowband Signal (Modulated Wave Packet):} In the time domain, the carrier shifts by phase delay $\tau_p$, while the envelope packet shifts by group delay $\tau_g$. In the frequency domain, the local phase spectrum around $\omega_0$ is linear with slope $-\tau_g(\omega_0)$.
    \item \textbf{Broadband Multi-Harmonic Signal (Linear Phase):} When $\tau_g(\omega) = \alpha$ (constant across all frequencies), transmission is \textbf{distortionless}. The entire composite signal is shifted by $\alpha$ samples in time with zero shape change.
    \item \textbf{Broadband Multi-Harmonic Signal (Non-Linear Phase):} When $\tau_g(\omega)$ varies with frequency, \textbf{phase dispersion} occurs. Harmonics shift asynchronously in time, resulting in severe waveform distortion, loss of symmetry, and peak amplitude attenuation.
\end{itemize}

\section{Impact on Voltage Signal Characteristics}
In power system harmonics, digital voltage buses, and analog instrumentation, non-constant group delay degrades key voltage parameters:

\begin{figure}[H]
\centering
\includegraphics[width=0.85\textwidth]{images/voltage_signal_dispersion.png}
\caption{Impact of linear vs. non-linear group delay dispersion on a multi-harmonic composite voltage waveform.}
\label{fig:voltage_dispersion}
\end{figure}

\begin{enumerate}
    \item \textbf{Peak Voltage Magnitude ($V_{peak}$) Degradation:} In multi-harmonic AC voltage signals $v(t) = \sum V_k \sin(k\omega_0 t + \phi_k)$, non-constant phase delays disrupt the constructive alignment of harmonic peaks, resulting in a reduced peak voltage amplitude or localized voltage overshoots.
    \item \textbf{Edge Slew Rate ($\frac{dV}{dt}$) \& Rise Time ($t_r$):} Sharp digital voltage steps suffer dispersion because high-frequency Fourier components are delayed differently from low frequencies. This smears the voltage step edge, reduces slew rate $\frac{dV}{dt}$, increases rise time $t_r$, and induces Inter-Symbol Interference (ISI) in data buses.
    \item \textbf{Crest Factor ($CF = V_{peak}/V_{rms}$):} Even for an all-pass filter where total signal energy and $V_{rms}$ are preserved, non-linear phase alters peak alignment, causing significant variations in the Crest Factor.
    \item \textbf{Zero-Crossing Jitter \& Synchronization Errors:} Non-uniform phase delay shifts zero-crossing timestamps of harmonic voltage components, introducing timing jitter in Zero-Crossing Detectors (ZCD), phase-locked loops (PLL), and SCR thyristor firing circuits.
\end{enumerate}

\section{Group Delay Derivation for First-Order Filter}
Let $y[n] - r y[n-1] = x[n]$ with $|r| < 1$.
\begin{align}
H(e^{j\omega}) &= \frac{1}{1 - r e^{-j\omega}} \\
\theta(\omega) &= -\arctan\left( \frac{r\sin\omega}{1 - r\cos\omega} \right)
\end{align}
Let $u(\omega) = \frac{r\sin\omega}{1 - r\cos\omega}$. Using the derivative of $\arctan(u)$:
\begin{equation}
\frac{du}{d\omega} = \frac{r\cos\omega - r^2}{(1-r\cos\omega)^2} \quad \text{and} \quad 1+u^2 = \frac{1+r^2-2r\cos\omega}{(1-r\cos\omega)^2}
\end{equation}
Combining these terms yields:
\begin{equation}
\frac{d\theta(\omega)}{d\omega} = -\frac{1}{1+u^2} \frac{du}{d\omega} = -\frac{r\cos\omega - r^2}{1 + r^2 - 2r\cos\omega}
\end{equation}
Thus:
\begin{equation}
\tau_g(\omega) = -\frac{d\theta(\omega)}{d\omega} = \frac{r\cos\omega - r^2}{1 + r^2 - 2r\cos\omega}
\end{equation}

\section{Checkpoint \& Numerical Review Questions}
\begin{enumerate}
    \item \textbf{Q1 (Pure Delay):} Find $\tau_p(\omega)$ and $\tau_g(\omega)$ for $h[n] = \delta[n - n_0]$. \\
    \textit{Solution:} $H(e^{j\omega}) = e^{-j\omega n_0} \implies \theta(\omega) = -\omega n_0$. Thus $\tau_p(\omega) = \tau_g(\omega) = n_0$ samples (constant delay $\to$ distortionless).
    \item \textbf{Q2 (DC Group Delay):} For $r = 0.9$, evaluate $\tau_g(0)$. \\
    \textit{Solution:} $\tau_g(0) = \frac{0.9 - 0.81}{1 + 0.81 - 1.8} = \frac{0.09}{0.01} = 9$ samples.
\end{enumerate}
