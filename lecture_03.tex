\section{Mathematical Definition of the DTFT}
The Discrete-Time Fourier Transform (DTFT) maps a discrete-time sequence $x[n]$ to a continuous, periodic complex frequency function $X(e^{j\omega})$:
\begin{equation}
X(e^{j\omega}) = \sum_{n=-\infty}^{\infty} x[n] e^{-j\omega n}
\end{equation}
\textit{Periodicity Proof:}
\begin{equation}
X(e^{j(\omega + 2\pi)}) = \sum_{n=-\infty}^{\infty} x[n] e^{-j(\omega + 2\pi)n} = \sum_{n=-\infty}^{\infty} x[n] e^{-j\omega n} e^{-j 2\pi n} = X(e^{j\omega})
\end{equation}
Since $e^{-j 2\pi n} = 1$ for any integer $n$, the transform is periodic with period $2\pi$.

The Inverse DTFT (IDTFT) is defined as:
\begin{equation}
x[n] = \frac{1}{2\pi} \int_{-\pi}^{\pi} X(e^{j\omega}) e^{j\omega n} d\omega
\end{equation}

\subsection{Convergence Conditions}
A sequence $x[n]$ has a uniformly convergent DTFT if it is absolutely summable:
\begin{equation}
\sum_{n=-\infty}^{\infty} |x[n]| < \infty
\end{equation}

\section{DTFT of Common Signals}
\subsection{Finite Rectangular Pulse (Length $M$)}
Let $x[n] = 1$ for $0 \le n \le M-1$, and $0$ otherwise.
\begin{equation}
X(e^{j\omega}) = \sum_{n=0}^{M-1} e^{-j\omega n} = \frac{1 - e^{-j\omega M}}{1 - e^{-j\omega}} = e^{-j\omega(M-1)/2} \frac{\sin\left(\frac{\omega M}{2}\right)}{\sin\left(\frac{\omega}{2}\right)}
\end{equation}
This is the periodic sinc function (Dirichlet kernel).

\subsection{Single-Sided Decaying Exponential}
Let $x[n] = a^n u[n]$ ($|a| < 1$).
\begin{equation}
X(e^{j\omega}) = \sum_{n=0}^{\infty} \left( a e^{-j\omega} \right)^n = \frac{1}{1 - a e^{-j\omega}}
\end{equation}
Magnitude:
\begin{equation}
|X(e^{j\omega})| = \frac{1}{\sqrt{1 + a^2 - 2a\cos\omega}}
\end{equation}

\subsection{Unit Impulse and Constant DC}
For $x[n] = \delta[n]$, $X(e^{j\omega}) = 1$.  
For $x[n] = 1$, $X(e^{j\omega}) = 2\pi \sum_{k=-\infty}^{\infty} \delta(\omega - 2\pi k)$.

\section{Key Properties of the DTFT}
\begin{itemize}
    \item \textbf{Linearity:} $a x[n] + b y[n] \leftrightarrow a X(e^{j\omega}) + b Y(e^{j\omega})$
    \item \textbf{Time Shifting:} $x[n - n_0] \leftrightarrow X(e^{j\omega}) e^{-j\omega n_0}$
    \item \textbf{Frequency Shifting (Modulation):} $x[n] e^{j\omega_0 n} \leftrightarrow X(e^{j(\omega - \omega_0)})$
    \item \textbf{Differentiation in Frequency:} $n x[n] \leftrightarrow j \frac{d X(e^{j\omega})}{d\omega}$
    \item \textbf{Convolution Theorem:} $x[n] * y[n] \leftrightarrow X(e^{j\omega}) \cdot Y(e^{j\omega})$
    \item \textbf{Parseval's Theorem:} $\sum_{n=-\infty}^{\infty} |x[n]|^2 = \frac{1}{2\pi} \int_{-\pi}^{\pi} |X(e^{j\omega})|^2 d\omega$
\end{itemize}

\section{Detailed Worked Numerical Examples}

\subsection{Example 1: DTFT of a Shifted Impulse Pair}
Find the DTFT of $x[n] = \delta[n] - \delta[n-1]$.
\begin{equation*}
X(e^{j\omega}) = 1 - e^{-j\omega} = e^{-j\omega/2} \left( e^{j\omega/2} - e^{-j\omega/2} \right) = 2j \sin(\omega/2) e^{-j\omega/2} = 2 \sin(\omega/2) e^{j(\pi/2 - \omega/2)}
\end{equation*}
Magnitude: $|X(e^{j\omega})| = 2|\sin(\omega/2)|$.

\subsection{Example 2: DTFT of a Modulated Signal}
Find the DTFT of $x[n] = a^n \cos(\omega_0 n) u[n]$ for $|a| < 1$.
Using Euler's formula:
\begin{equation*}
X(e^{j\omega}) = \text{DTFT}\left\{ \frac{1}{2} a^n e^{j\omega_0 n} u[n] + \frac{1}{2} a^n e^{-j\omega_0 n} u[n] \right\} = \frac{1}{2} \left[ \frac{1}{1 - a e^{-j(\omega - \omega_0)}} + \frac{1}{1 - a e^{-j(\omega + \omega_0)}} \right]
\end{equation*}

\subsection{Example 3: Energy Calculation using Parseval's}
Given $x[n] = 0.5^n u[n]$, calculate total energy.
In the time domain:
\begin{equation*}
E = \sum_{n=0}^{\infty} (0.25)^n = \frac{1}{1 - 0.25} = \frac{4}{3}
\end{equation*}
In the frequency domain using Parseval's formula:
\begin{equation*}
E = \frac{1}{2\pi} \int_{-\pi}^{\pi} \frac{1}{1.25 - \cos\omega} d\omega = \frac{1}{\sqrt{1.25^2 - 1}} = \frac{1}{\sqrt{1.5625 - 1}} = \frac{1}{\sqrt{0.5625}} = \frac{1}{0.75} = \frac{4}{3}
\end{equation*}
Both domains yield identical energy, verifying Parseval's theorem.
