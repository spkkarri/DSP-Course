\section{Overview of the Inverse Z-Transform}
The Inverse Z-transform maps $X(z)$ and its ROC back to a discrete sequence $x[n]$:
\begin{equation}
x[n] = \mathcal{Z}^{-1}\{X(z)\}
\end{equation}
Specification of the ROC is critical to ensure uniqueness.

\section{The Three Inversion Methods}
\begin{enumerate}
    \item \textbf{Partial Fraction Expansion (PFE):} Decomposes $\frac{X(z)}{z}$ into first/second-order pole components:
    \begin{equation}
    \frac{X(z)}{z} = \sum_{k=1}^{N} \frac{A_k}{z-p_k}, \qquad A_k = \left[ (z-p_k) \frac{X(z)}{z} \right]_{z=p_k}
    \end{equation}
    \item \textbf{Power Series Expansion (Long Division):} Direct polynomial long division:
    \begin{equation}
    X(z) = \sum_{n=-\infty}^{\infty} x[n] z^{-n} = \dots + x[-1]z^1 + x[0] + x[1]z^{-1} + x[2]z^{-2} + \dots
    \end{equation}
    \item \textbf{Contour Integration (Cauchy's Residue Theorem):}
    \begin{equation}
    x[n] = \frac{1}{2\pi j} \oint_{C} X(z) z^{n-1} dz = \sum \text{Residues of } \left[ X(z)z^{n-1} \right] \text{ inside } C
    \end{equation}
\end{enumerate}

\section{Worked Numerical 1: Comparing All 3 Inversion Approaches}
Given $X(z) = \frac{1}{1 - 1.5 z^{-1} + 0.5 z^{-2}} = \frac{z^2}{(z - 1)(z - 0.5)}$:
\begin{itemize}
    \item \textbf{PFE Decomposition:} $\frac{X(z)}{z} = \frac{2}{z-1} - \frac{1}{z-0.5} \implies X(z) = \frac{2}{1-z^{-1}} - \frac{1}{1-0.5 z^{-1}}$.
    \item \textbf{Causal Sequence ($|z|>1$):} $x[n] = \left[ 2(1)^n - (0.5)^n \right] u[n] \implies \{1.0, 1.5, 1.75, 1.875, \dots\}$.
    \item \textbf{Anti-Causal Sequence ($|z|<0.5$):} $x[n] = \left[ (0.5)^n - 2 \right] u[-n-1] \implies \{\dots, 14, 6, 2, 0\}$.
    \item \textbf{Two-Sided Sequence ($0.5<|z|<1$):} $x[n] = -(0.5)^n u[n] - 2 u[-n-1]$.
\end{itemize}
All three methods (PFE, polynomial long division, and residue integration) yield identical analytical formulas and sample values.

\section{Worked Numerical 2: Complex Conjugate Poles (Sinusoids \& Damped Cosines)}
Given $X(z) = \frac{z^2 - \frac{\sqrt{2}}{2} z}{z^2 - \sqrt{2} z + 1}$ with causal ROC $|z|>1$:
The poles lie on the unit circle at $p_{1,2} = e^{\pm j\pi/4}$ ($r=1, \omega_0 = \pi/4$).
Using the standard cosine pair $\mathcal{Z}\{r^n \cos(\omega_0 n) u[n]\} = \frac{z(z - r\cos\omega_0)}{z^2 - 2r\cos\omega_0 z + r^2}$:
\begin{equation}
x[n] = \cos\left(\frac{\pi}{4} n\right) u[n] \implies \{1.0, 0.707, 0.0, -0.707, -1.0, \dots\}
\end{equation}

\section{Worked Numerical 3: Repeated Poles ($a^n$ and $n a^n$ Ramp Modes)}
Given $X(z) = \frac{z^2}{(z - 0.5)^2}$ with causal ROC $|z|>0.5$:
\begin{equation}
\frac{X(z)}{z} = \frac{z}{(z-0.5)^2} = \frac{1.0}{z-0.5} + \frac{0.5}{(z-0.5)^2}
\end{equation}
Multiplying by $z$ and applying inverse pairs $\frac{z}{z-a} \leftrightarrow a^n u[n]$ and $\frac{a z}{(z-a)^2} \leftrightarrow n a^n u[n]$:
\begin{equation}
x[n] = (0.5)^n u[n] + n(0.5)^n u[n] = (n+1)(0.5)^n u[n]
\end{equation}

\section{Convolution Theorem Equivalence: Time-Domain Sum $\leftrightarrow$ Z-Domain Product}

\begin{figure}[H]
    \centering
    \includegraphics[width=0.85\textwidth]{images/convolution_time_vs_z_domain.png}
    \caption{Convolution Property Equivalence: Time-Domain Folding vs. Z-Domain Algebraic Multiplication.}
\end{figure}

Given input $x[n] = (0.5)^n u[n]$ and impulse response $h[n] = (0.8)^n u[n]$:
\begin{itemize}
    \item \textbf{Time-Domain Sum:} $y[n] = \sum_{k=0}^n (0.5)^k (0.8)^{n-k} = \frac{8}{3}(0.8)^n - \frac{5}{3}(0.5)^n, \; n \ge 0$.
    \item \textbf{Z-Domain Product:} $Y(z) = X(z) H(z) = \frac{z^2}{(z-0.5)(z-0.8)} = \frac{8/3 z}{z-0.8} - \frac{5/3 z}{z-0.5}$.
    \item \textbf{Inverse Z Output:} $y[n] = \left[ \frac{8}{3}(0.8)^n - \frac{5}{3}(0.5)^n \right] u[n]$.
\end{itemize}
The Z-domain method replaces complex piecewise sliding summations with simple polynomial multiplication.

\subsection{Worked Problem with Initial Conditions}
Solve $y[n] - 0.7 y[n-1] + 0.1 y[n-2] = x[n]$ for $n \ge 0$, with input $x[n] = u[n]$ and initial conditions $y[-1]=2, y[-2]=1$.

\noindent \textbf{Step 1: Unilateral Z-Transform Property}
\begin{equation}
Y(z) (1 - 0.7 z^{-1} + 0.1 z^{-2}) - 0.7(2) + 0.1(2 z^{-1} + 1) = X(z)
\end{equation}
\begin{equation}
Y(z) = \underbrace{\frac{1.3 - 0.2 z^{-1}}{1 - 0.7 z^{-1} + 0.1 z^{-2}}}_{Y_{zi}(z) \text{ (Zero-Input Response)}} + \underbrace{\frac{1}{1 - 0.7 z^{-1} + 0.1 z^{-2}} X(z)}_{Y_{zs}(z) \text{ (Zero-State Response)}}
\end{equation}

\noindent \textbf{Step 2: Zero-Input Response}
\begin{equation}
Y_{zi}(z) = \frac{1.5 z}{z-0.5} - \frac{0.2 z}{z-0.2} \implies y_{zi}[n] = \left[ 1.5(0.5)^n - 0.2(0.2)^n \right] u[n]
\end{equation}

\noindent \textbf{Step 3: Zero-State Response with $X(z) = \frac{z}{z-1}$}
\begin{equation}
Y_{zs}(z) = \frac{2.5 z}{z-1} - \frac{5/3 z}{z-0.5} + \frac{1/6 z}{z-0.2} \implies y_{zs}[n] = \left[ 2.5 - \frac{5}{3}(0.5)^n + \frac{1}{6}(0.2)^n \right] u[n]
\end{equation}

\noindent \textbf{Step 4: Total Response $y[n] = y_{zi}[n] + y_{zs}[n]$}
\begin{equation}
\mathbf{y[n] = \left[ 2.5 - \frac{1}{6}(0.5)^n - \frac{1}{30}(0.2)^n \right] u[n]}
\end{equation}
Verified: $y[0] = 2.3$ and $y[1] = 2.41$, matching difference recursion exactly.
