\section*{2. MATHEMATICAL FOUNDATIONS}

\subsection*{2.1 FIR Lattice Recursion \& Step-Down Algorithm}
\begin{itemize}
\item \textbf{Lattice Recursion:}
\end{itemize}
  $$ A_m(z) = A_{m-1}(z) + k_m z^{-1} B_{m-1}(z) $$
  $$ B_m(z) = z^{-m} A_m(z^{-1}) \quad (\text{Reverse Polynomial}) $$
\begin{itemize}
\item \textbf{Step-Down Algorithm (Direct $\to$ Lattice):}
\end{itemize}
  Given $A_N(z) = 1 + \sum_{i=1}^N \alpha_N(i) z^{-i}$:
  1. Set $k_N = \alpha_N(N)$.
  2. For $m = N, N-1, \dots, 2$:
     $$ \alpha_{m-1}(i) = \frac{\alpha_m(i) - k_m \alpha_m(m - i)}{1 - k_m^2}, \quad i = 1, 2, \dots, m-1 $$
     $$ k_{m-1} = \alpha_{m-1}(m-1) $$

\section*{3. WORKED NUMERICAL EXAMPLES}

\subsection*{Example 20.1: Step-Down Conversion to Reflection Coefficients}
\textbf{Problem:} Find the reflection coefficients $k_1, k_2, k_3$ for the FIR filter:
$$ A_3(z) = 1 - 0.9 z^{-1} + 0.64 z^{-2} - 0.5 z^{-3} $$

\textbf{Solution:}
\begin{itemize}
\item \textbf{Stage 3:}
\end{itemize}
  $k_3 = \alpha_3(3) = -0.5$.
  $1 - k_3^2 = 1 - (-0.5)^2 = 1 - 0.25 = 0.75$.
\begin{itemize}
\item \textbf{Compute $\alpha_2(i)$ for $m=3$:}
\end{itemize}
  * $\alpha_2(1) = \frac{\alpha_3(1) - k_3 \alpha_3(2)}{0.75} = \frac{-0.9 - (-0.5)(0.64)}{0.75} = \frac{-0.9 + 0.32}{0.75} = \frac{-0.58}{0.75} = -0.7733$.
  * $\alpha_2(2) = \frac{\alpha_3(2) - k_3 \alpha_3(1)}{0.75} = \frac{0.64 - (-0.5)(-0.9)}{0.75} = \frac{0.64 - 0.45}{0.75} = \frac{0.19}{0.75} = 0.2533$.
  $k_2 = \alpha_2(2) = 0.2533$.
\begin{itemize}
\item \textbf{Stage 2:}
\end{itemize}
  $1 - k_2^2 = 1 - (0.2533)^2 = 1 - 0.0642 = 0.9358$.
\begin{itemize}
\item \textbf{Compute $\alpha_1(i)$ for $m=2$:}
\end{itemize}
  * $\alpha_1(1) = \frac{\alpha_2(1) - k_2 \alpha_2(1)}{0.9358} = \frac{-0.7733(1 - 0.2533)}{0.9358} = \frac{-0.7733 \times 0.7467}{0.9358} = -0.6170$.
  $k_1 = \alpha_1(1) = -0.6170$.
\begin{itemize}
\item \textbf{Stability Check:}
\end{itemize}
  $|k_1| = 0.6170 < 1, \; |k_2| = 0.2533 < 1, \; |k_3| = 0.5000 < 1$.
  All $|k_m| < 1 \implies$ The corresponding all-pole filter is \textbf{Stable}.
