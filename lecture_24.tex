\section*{2. MATHEMATICAL FOUNDATIONS}

\subsection*{2.1 Frequency-Sampling Design Equations}
Let $H[k] = H_d(e^{j \frac{2\pi k}{N}})$. For a linear-phase FIR filter of length $N$:
$$ H[k] = |H[k]| e^{j \angle H[k]}, \quad \angle H[k] = -\left( \frac{N-1}{2} \right) \frac{2\pi k}{N} $$
The impulse response is computed by the IDFT:
$$ h[n] = \frac{1}{N} \sum_{k=0}^{N-1} H[k] e^{j \frac{2\pi k n}{N}} $$
Exploiting Hermitian symmetry ($H[N-k] = H^*[k]$):
\begin{itemize}
\item \textbf{For $N$ Odd:}
\end{itemize}
  $$ h[n] = \frac{H[0]}{N} + \frac{2}{N} \sum_{k=1}^{(N-1)/2} |H[k]| \cos\left[ \frac{2\pi k}{N} \left( n - \frac{N-1}{2} \right) \right] $$
\begin{itemize}
\item \textbf{For $N$ Even:}
\end{itemize}
  $$ h[n] = \frac{H[0]}{N} + \frac{2}{N} \sum_{k=1}^{N/2 - 1} |H[k]| \cos\left[ \frac{2\pi k}{N} \left( n - \frac{N-1}{2} \right) \right] + \frac{H[N/2]}{N} (-1)^n $$

\section*{3. WORKED NUMERICAL EXAMPLES}

\subsection*{Example 24.1: Frequency-Sampling Lowpass Filter Design}
\textbf{Problem:} Design a 7-point ($N=7$) linear-phase FIR lowpass filter with frequency samples:
$$ |H[k]| = 1, 1, 0, 0, 0, 0, 1 \quad \text{for } k = 0, 1, 2, 3, 4, 5, 6 $$

\textbf{Solution:}
\begin{itemize}
\item Length $N = 7$ (Odd), $\tau = \frac{7-1}{2} = 3$.
\item Non-zero samples: $|H[0]| = 1; \; |H[1]| = |H[6]| = 1$.
\item Using the real summation formula:
  $$ h[n] = \frac{H[0]}{7} + \frac{2}{7} |H[1]| \cos\left[ \frac{2\pi (1)}{7} (n - 3) \right] = \frac{1}{7} + \frac{2}{7} \cos\left[ \frac{2\pi}{7} (n - 3) \right] $$
  \begin{itemize}
  \item $h[3] = \frac{1}{7} + \frac{2}{7} \cos(0) = \frac{3}{7} \approx 0.4286$
  \item $h[2] = h[4] = \frac{1}{7} + \frac{2}{7} \cos\left( \frac{2\pi}{7} \right) = \frac{1 + 2(0.6235)}{7} = \frac{2.2470}{7} \approx 0.3210$
  \item $h[1] = h[5] = \frac{1}{7} + \frac{2}{7} \cos\left( \frac{4\pi}{7} \right) = \frac{1 + 2(-0.2225)}{7} = \frac{0.5550}{7} \approx 0.0793$
  \item $h[0] = h[6] = \frac{1}{7} + \frac{2}{7} \cos\left( \frac{6\pi}{7} \right) = \frac{1 + 2(-0.9010)}{7} = \frac{-0.8019}{7} \approx -0.1146$
  \end{itemize}
\end{itemize}
$$ h[n] = \underset{\uparrow}{-0.1146}, 0.0793, 0.3210, 0.4286, 0.3210, 0.0793, -0.1146 $$
