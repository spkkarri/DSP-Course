\section{2. MOTIVATION: DESIGNING IIR FILTERS FROM ANALOG PROTOTYPES}

\subsection{2.1 FIR vs. IIR Filtering Paradigms}
In Lectures 21--23, we designed Finite Impulse Response (FIR) filters. FIR filters possess exact linear phase and guaranteed stability. However, achieving sharp transition roll-offs requires very high filter orders ($N = 50$ to $500$), incurring substantial memory and arithmetic latency.

\textbf{Infinite Impulse Response (IIR) filters}, governed by the rational transfer function:
\begin{equation}
H(z) = \frac{\sum_{k=0}^M b_k z^{-k}}{1 + \sum_{k=1}^N a_k z^{-k}}
\end{equation}
exploit recursive feedback (poles). Consequently, an IIR filter can meet extremely stringent magnitude specifications with a fraction of the filter order ($N = 2$ to $8$) compared to an FIR filter.

\subsection{2.2 Why Start from Continuous-Time Analog Filters?}
Continuous-time analog filter theory (Butterworth, Chebyshev, Elliptic) is a mature, mathematically closed engineering discipline developed over a century. Closed-form polynomials exist that yield optimal, mathematically proven approximations to ideal brick-wall responses.

Therefore, the standard strategy for designing a digital IIR filter is:
\begin{enumerate}
    \item Translate digital frequency specifications ($\omega_p, \omega_s, \delta_p, \delta_s$) into equivalent analog specifications ($\Omega_p, \Omega_s$).
    \item Design an analog prototype filter $H_a(s)$ meeting those specifications.
    \item Transform the continuous-time system function $H_a(s)$ into a causal discrete-time system function $H(z)$ via an algebraic or time-domain mapping.
\end{enumerate}

\noindent\textbf{Core Requirements for the Mapping Technique:}
\begin{itemize}
    \item \textbf{Preservation of Stability:} Poles in the stable left-half $s$-plane ($\text{Re}(s) < 0$) \textbf{MUST} map inside the unit circle of the $z$-plane ($|z| < 1$).
    \item \textbf{Preservation of Frequency Characteristics:} The imaginary axis of the $s$-plane ($s = j\Omega$) must map onto the unit circle ($z = e^{j\omega}$) so that analog passbands and stopbands translate to digital passbands and stopbands.
\end{itemize}

\section{3. THE IMPULSE INVARIANCE METHOD: MATHEMATICAL FOUNDATIONS}

\subsection{3.1 The Time-Domain Invariance Criterion}
The philosophy of the \textbf{Impulse Invariance Method} is to produce a digital filter whose discrete impulse response $h[n]$ is an exact sampled replica of the continuous-time analog impulse response $h_a(t)$:
\begin{equation}
\mathbf{h[n] = T_s \, h_a(n T_s)}, \quad n = 0, 1, 2, \dots
\label{eq:impulse_def}
\end{equation}
where $T_s$ is the sampling interval (seconds). The scaling factor $T_s$ is introduced to ensure that the digital filter's DC gain matches the analog filter's DC gain:
\begin{equation}
H(e^{j0}) = \sum_{n=0}^\infty h[n] = T_s \sum_{n=0}^\infty h_a(n T_s) \approx \int_0^\infty h_a(t) dt = H_a(0)
\end{equation}

\subsection{3.2 Mapping of Poles from s-Plane to z-Plane}
Let the analog filter transfer function $H_a(s)$ be described by distinct poles $s_k$:
\begin{equation}
H_a(s) = \sum_{k=1}^N \frac{A_k}{s - s_k}
\label{eq:ha_pfe}
\end{equation}
Taking the inverse Laplace transform of each simple pole term yields the causal analog impulse response:
\begin{equation}
h_a(t) = \sum_{k=1}^N A_k e^{s_k t} u(t)
\end{equation}
Sampling this response at $t = n T_s$ and multiplying by $T_s$ as per \eqref{eq:impulse_def}:
\begin{equation}
h[n] = T_s \sum_{k=1}^N A_k e^{s_k n T_s} u[n] = \sum_{k=1}^N \big(T_s A_k\big) \left(e^{s_k T_s}\right)^n u[n]
\end{equation}
Now, taking the unilateral $z$-transform of $h[n]$:
\begin{align}
H(z) &= \sum_{n=0}^\infty h[n] z^{-n} = \sum_{n=0}^\infty \left[ \sum_{k=1}^N T_s A_k \big(e^{s_k T_s}\big)^n \right] z^{-n} \nonumber \\
     &= \sum_{k=1}^N T_s A_k \sum_{n=0}^\infty \left( e^{s_k T_s} z^{-1} \right)^n
\end{align}
Applying the standard geometric series summation $\sum_{n=0}^\infty a^n = \frac{1}{1 - a}$ (convergent for $|e^{s_k T_s} z^{-1}| < 1$):
\begin{equation}
\mathbf{H(z) = \sum_{k=1}^N \frac{T_s A_k}{1 - e^{s_k T_s} z^{-1}}}
\label{eq:hz_pfe}
\end{equation}

\noindent\textbf{Fundamental Operational Rule of Impulse Invariance:}
\begin{equation}
\mathbf{\frac{A_k}{s - s_k} \quad \xrightarrow{\quad \text{Impulse Invariance} \quad} \quad \frac{T_s A_k}{1 - e^{s_k T_s} z^{-1}}}
\end{equation}
Every analog pole at $s = s_k$ maps directly to a discrete digital pole at:
\begin{equation}
\mathbf{z_k = e^{s_k T_s}}
\end{equation}

\subsection{3.3 Stability Mapping Analysis}
Let an analog pole be written in rectangular form: $s_k = \sigma_k + j\Omega_k$.
The corresponding digital pole is:
\begin{equation}
z_k = e^{(\sigma_k + j\Omega_k)T_s} = e^{\sigma_k T_s} \cdot e^{j \Omega_k T_s} \implies |z_k| = e^{\sigma_k T_s}
\end{equation}
\begin{enumerate}
    \item \textbf{Stable Analog Poles ($\sigma_k < 0$):} If the analog pole lies in the left-half of the $s$-plane, then $\sigma_k < 0 \implies |z_k| = e^{\sigma_k T_s} < 1$. The pole maps \textbf{strictly inside the unit circle}!
    \item \textbf{Marginally Stable Poles ($\sigma_k = 0$):} If the analog pole lies on the $j\Omega$-axis, $|z_k| = e^0 = 1$. The pole maps \textbf{on the unit circle}.
    \item \textbf{Unstable Poles ($\sigma_k > 0$):} If the analog pole lies in the right-half plane, $|z_k| = e^{\sigma_k T_s} > 1$. The pole maps \textbf{outside the unit circle}.
\end{enumerate}
\textbf{Theorem:} The Impulse Invariance method \textbf{unconditionally preserves system stability}. A stable continuous-time filter always yields a strictly stable discrete-time digital filter.

\section{4. THE FREQUENCY ALIASING PHENOMENON}

\subsection{4.1 Frequency-Domain Mathematical Relationship}
Because $h[n]$ is obtained by sampling $h_a(t)$ at uniform intervals $T_s$, the continuous-time Fourier transform $H_a(j\Omega)$ and the discrete-time Fourier transform $H(e^{j\omega})$ are related by the standard Poisson Summation formula:
\begin{equation}
\mathbf{H(e^{j\omega}) = \sum_{k=-\infty}^\infty H_a\left( j \frac{\omega - 2\pi k}{T_s} \right)}
\label{eq:aliasing_sum}
\end{equation}
where normalized digital frequency $\omega$ is related to analog angular frequency $\Omega$ by:
\begin{equation}
\omega = \Omega T_s
\end{equation}

\subsection{4.2 Origin of Severe Aliasing Distortion}
Equation \eqref{eq:aliasing_sum} shows that $H(e^{j\omega})$ is an infinite periodic superposition of replicas of $H_a(j\Omega)$ shifted by integer multiples of the sampling frequency $\Omega_s = \frac{2\pi}{T_s}$.

Unlike discrete signals which can be passed through continuous anti-aliasing lowpass filters before digitization, \textbf{an analog filter transfer function $H_a(s)$ is a continuous mathematical rational function whose frequency response extends to $\Omega \to \pm\infty$}. 
\begin{itemize}
    \item Because $H_a(j\Omega)$ cannot be strictly band-limited to $|\Omega| \le \frac{\pi}{T_s}$, the high-frequency tails fold back into the baseband $[-\pi, \pi]$!
    \item This spectral foldover causes **irreversible aliasing distortion** in the passband and transition band.
\end{itemize}

\subsection{4.3 Why Impulse Invariance Fails for Highpass and Bandstop Filters}
For an analog Highpass filter, $|H_a(j\Omega)| \to 1$ as $\Omega \to \infty$. 
When transformed via impulse invariance:
\begin{itemize}
    \item The non-zero high-frequency response aliases across all periodic copies $k = \pm 1, \pm 2, \dots$.
    \item These infinite overlapping tails superimpose at DC ($\omega = 0$), causing the digital filter's response to be non-zero at DC!
    \item \textbf{Fatal Consequence:} The designed filter completely fails to attenuate low frequencies.
\end{itemize}
\textbf{Golden Rule:} The Impulse Invariance method \textbf{CANNOT be used to design Highpass, Bandstop, or Differentiator filters}. It is strictly suitable only for **band-limited Lowpass and Bandpass filters** where the stopband attenuation at Nyquist ($\Omega = \pi/T_s$) is already sufficiently deep ($> 30\text{--}40$ dB).

\section{5. SYSTEMATIC STEP-BY-STEP DESIGN PROTOCOL}

To design a digital IIR lowpass filter from continuous specifications:
\begin{enumerate}
    \item \textbf{Map Digital Frequencies to Analog:} Using $\Omega = \omega / T_s$, compute:
    \begin{equation}
    \Omega_p = \frac{\omega_p}{T_s}, \quad \Omega_s = \frac{\omega_s}{T_s}
    \end{equation}
    \item \textbf{Determine Analog Filter Order $N$ and Cutoff $\Omega_c$:} For a Butterworth prototype with passband ripple $A_p$ (dB) and stopband attenuation $A_s$ (dB):
    \begin{equation}
    N \ge \frac{\log_{10}\left( \frac{10^{0.1 A_s} - 1}{10^{0.1 A_p} - 1} \right)}{2\log_{10}(\Omega_s / \Omega_p)}, \quad \Omega_c = \frac{\Omega_p}{\left(10^{0.1 A_p} - 1\right)^{1/(2N)}}
    \end{equation}
    \item \textbf{Formulate Analog Transfer Function $H_a(s)$:} Place $N$ poles symmetrically on the left-half circle of radius $\Omega_c$:
    \begin{equation}
    s_k = \Omega_c e^{j \frac{\pi}{2N}(2k + N - 1)}, \quad k = 1, 2, \dots, N
    \end{equation}
    \item \textbf{Perform Partial Fraction Expansion:} Express $H_a(s)$ as a sum of simple 1st-order terms:
    \begin{equation}
    H_a(s) = \sum_{k=1}^N \frac{A_k}{s - s_k}, \quad \text{where } A_k = \lim_{s \to s_k} (s - s_k) H_a(s)
    \end{equation}
    \item \textbf{Transform to Digital $H(z)$:} Replace each analog partial fraction using the mapping rule:
    \begin{equation}
    H(z) = \sum_{k=1}^N \frac{T_s A_k}{1 - e^{s_k T_s} z^{-1}}
    \end{equation}
    \item \textbf{Combine Conjugate Pairs into Real Biquads:} Combine complex conjugate terms to obtain strictly real denominator polynomials.
\end{enumerate}

\section{6. STEP-BY-STEP WORKED NUMERICAL EXAMPLES}

\subsection{Example 24.1: First-Order Analog to Digital IIR Transformation}
\textbf{Problem Statement:} An analog lowpass filter is described by the transfer function:
\begin{equation}
H_a(s) = \frac{2}{s + 2}
\end{equation}
Determine the digital filter transfer function $H(z)$ using the Impulse Invariance method with a sampling period of $T_s = 0.5$ seconds.

\vspace{1mm}
\noindent\textbf{Solution:}

\noindent\textbf{Step 1: Identify Pole and Residue}
$H_a(s)$ has a single real pole at $s_1 = -2$ with residue $A_1 = 2$.

\noindent\textbf{Step 2: Apply the Impulse Invariance Mapping Rule}
Using \eqref{eq:hz_pfe} with $T_s = 0.5$ s:
\begin{equation}
H(z) = \frac{T_s A_1}{1 - e^{s_1 T_s} z^{-1}}
\end{equation}
Substituting $A_1 = 2$, $s_1 = -2$, and $T_s = 0.5$:
\begin{equation}
s_1 T_s = (-2)(0.5) = -1.0
\end{equation}
\begin{equation}
T_s A_1 = (0.5)(2) = 1.0
\end{equation}
\begin{equation}
e^{s_1 T_s} = e^{-1} \approx 0.367879
\end{equation}

\noindent\textbf{Step 3: Formulate Digital Transfer Function}
\begin{equation}
\mathbf{H(z) = \frac{1}{1 - 0.3679 z^{-1}}}
\end{equation}

\noindent\textbf{Step 4: DC Gain Verification}
\begin{itemize}
    \item Analog DC Gain: $H_a(0) = \frac{2}{0 + 2} = 1.0$.
    \item Digital DC Gain: At $z = 1$:
    \begin{equation}
    H(1) = \frac{1}{1 - 0.3679} = \frac{1}{0.6321} \approx \mathbf{1.5819}
    \end{equation}
    Notice that $H(1) \ne 1.0$ due to the discrete approximation of continuous integration by rectangular summation! If exact DC matching is desired, the gain is scaled by $(1 - e^{-1}) \approx 0.6321$.
\end{itemize}

\vspace{2mm}
\subsection{Example 24.2: Second-Order Butterworth Filter Design with Complex Poles}
\textbf{Problem Statement:} An analog second-order Butterworth lowpass filter with cutoff $\Omega_c = 1$ rad/s has the transfer function:
\begin{equation}
H_a(s) = \frac{1}{s^2 + \sqrt{2} s + 1}
\end{equation}
Design the corresponding digital IIR filter using impulse invariance with $T_s = 1$ second.

\vspace{1mm}
\noindent\textbf{Solution:}

\noindent\textbf{Step 1: Factor the Denominator to Find Analog Poles}
\begin{equation}
s^2 + \sqrt{2} s + 1 = 0 \implies s_{1,2} = \frac{-\sqrt{2} \pm \sqrt{2 - 4}}{2} = -\frac{1}{\sqrt{2}} \pm j \frac{1}{\sqrt{2}} \approx -0.7071 \pm j 0.7071
\end{equation}

\noindent\textbf{Step 2: Partial Fraction Expansion}
\begin{equation}
H_a(s) = \frac{1}{(s - s_1)(s - s_2)} = \frac{A_1}{s - s_1} + \frac{A_2}{s - s_2}
\end{equation}
where:
\begin{equation}
A_1 = \left. \frac{1}{s - s_2} \right|_{s = s_1} = \frac{1}{s_1 - s_2} = \frac{1}{j 2(0.7071)} = \frac{1}{j \sqrt{2}} = -j \frac{1}{\sqrt{2}} \approx -j 0.7071
\end{equation}
By complex conjugate symmetry, $A_2 = A_1^* = j \frac{1}{\sqrt{2}} \approx +j 0.7071$.

\noindent\textbf{Step 3: Map Poles to the Z-Domain}
With $T_s = 1$ s:
\begin{equation}
z_1 = e^{s_1 T_s} = e^{-0.7071 + j 0.7071} = e^{-0.7071} \big[\cos(0.7071 \text{ rad}) + j \sin(0.7071 \text{ rad})\big]
\end{equation}
Noting that $0.7071 \text{ rad} = 0.7071 \times \frac{180^\circ}{\pi} \approx 40.51^\circ$:
\begin{equation}
e^{-0.7071} \approx 0.49307
\end{equation}
\begin{equation}
\cos(40.51^\circ) \approx 0.76024, \quad \sin(40.51^\circ) \approx 0.64964
\end{equation}
\begin{equation}
z_1 = 0.49307(0.76024 + j 0.64964) \approx \mathbf{0.3748 + j 0.3203}
\end{equation}
and $z_2 = z_1^* = \mathbf{0.3748 - j 0.3203}$.
Magnitude: $|z_1| = e^{-0.7071} = 0.4931 < 1$ ($\implies$ strictly stable!).

\noindent\textbf{Step 4: Formulate Digital Partial Fractions}
\begin{equation}
H(z) = \frac{T_s A_1}{1 - z_1 z^{-1}} + \frac{T_s A_2}{1 - z_2 z^{-1}}
\end{equation}
Substituting $T_s = 1$, $A_1 = -j/\sqrt{2}$, $A_2 = +j/\sqrt{2}$:
\begin{equation}
H(z) = \frac{-j \frac{1}{\sqrt{2}}}{1 - z_1 z^{-1}} + \frac{j \frac{1}{\sqrt{2}}}{1 - z_1^* z^{-1}}
\end{equation}
Putting over a common denominator:
\begin{align}
H(z) &= \frac{-j \frac{1}{\sqrt{2}}(1 - z_1^* z^{-1}) + j \frac{1}{\sqrt{2}}(1 - z_1 z^{-1})}{(1 - z_1 z^{-1})(1 - z_1^* z^{-1})} \nonumber \\
     &= \frac{j \frac{1}{\sqrt{2}}(z_1^* - z_1) z^{-1}}{1 - 2\text{Re}(z_1) z^{-1} + |z_1|^2 z^{-2}}
\end{align}
Evaluating the numerator:
\begin{equation}
z_1^* - z_1 = -j 2\text{Im}(z_1) = -j 2(0.3203) = -j 0.6406
\end{equation}
\begin{equation}
j \frac{1}{\sqrt{2}}(z_1^* - z_1) = j \frac{1}{\sqrt{2}}(-j 0.6406) = \frac{0.6406}{\sqrt{2}} \approx \mathbf{0.4530}
\end{equation}
Evaluating the denominator:
\begin{equation}
2\text{Re}(z_1) = 2(0.3748) = \mathbf{0.7496}
\end{equation}
\begin{equation}
|z_1|^2 = (0.49307)^2 \approx \mathbf{0.2431}
\end{equation}

\noindent\textbf{Step 5: Final Real Digital Biquad Transfer Function}
\begin{equation}
\mathbf{H(z) = \frac{0.4530 z^{-1}}{1 - 0.7496 z^{-1} + 0.2431 z^{-2}}}
\end{equation}
Notice that all coefficients are strictly real, and the poles lie comfortably inside the unit circle at radius $r = 0.4931$.

\section{7. FACULTY INSIGHTS, COMMON PITFALLS \& EXAM TRAPS}

\begin{itemize}
    \item \textbf{Pitfall 1: Applying Impulse Invariance to the Whole Function Instead of Partial Fractions:}
    Students frequently try to substitute $s = \frac{1}{T_s}\ln(z)$ directly into $H_a(s)$. This produces a transcendental function, not a rational polynomial! Impulse invariance \textbf{must always be performed on partial fraction expansions}: $\frac{A}{s - s_k} \to \frac{T A}{1 - e^{s_k T} z^{-1}}$.
    
    \item \textbf{Pitfall 2: Forgetting the Scaling Factor $T_s$:}
    Omitting $T_s$ in the numerator produces a filter whose gain is magnified by $1/T_s$. For $T_s = 0.001$ s, the gain will be $1000\times$ too high!
    
    \item \textbf{Pitfall 3: Believing Zero Locations Map via $z = e^{s T_s}$:}
    \textbf{Only poles map via $z_k = e^{s_k T_s}$. ZEROS DO NOT MAP via $z = e^{s T_s}$!} In Example 24.2, $H_a(s)$ had no finite zeros, yet $H(z)$ developed a zero at $z = 0$.
    
    \item \textbf{Pitfall 4: Attempting to Design a Highpass Filter via Impulse Invariance:}
    In semester exams, questions sometimes ask students to design a Highpass filter via Impulse Invariance. The astute student must state immediately that Impulse Invariance is fundamentally invalid for Highpass design due to catastrophic aliasing at DC.
    
    \item \textbf{Pitfall 5: Angle Unit Confusion in Handheld Calculators:}
    When computing $e^{s_k T_s} = e^{\sigma T_s} e^{j \Omega T_s}$, $\Omega T_s$ is in **radians**, not degrees! Plugging radians into a calculator set to degree mode causes massive coefficient errors.
\end{itemize}
