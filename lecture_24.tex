\section{2. THE FREQUENCY SAMPLING PARADIGM: MATHEMATICAL FOUNDATIONS}

\subsection{2.1 Motivation: Direct Frequency-Domain Filter Specification}
In Lectures 22 and 23, we designed FIR filters using the **windowing method**, which begins in the continuous frequency domain ($H_d(e^{j\omega})$), transforms via inverse DTFT to the continuous infinite impulse response ($h_d[n]$), and truncates using a window sequence $w[n]$. However, the windowing method exhibits key limitations:
\begin{itemize}
    \item Inverse DTFT integrals $\int_{-\pi}^{\pi} H_d(e^{j\omega}) e^{j\omega n} d\omega$ can rarely be evaluated analytically for non-standard, arbitrary, or experimentally measured frequency responses.
    \item The designer has no direct control over the exact values of the frequency response at specific discrete frequency bins of interest.
\end{itemize}
The \textbf{Frequency Sampling Method} provides an elegant, highly flexible alternative: rather than approximating an infinite time-domain sinc function, we **directly specify the desired filter response at $N$ uniformly spaced frequency points on the unit circle**:
\begin{equation}
\omega_k = \frac{2\pi k}{N}, \quad k = 0, 1, \dots, N-1
\label{eq:grid_frequencies}
\end{equation}
The discrete frequency samples $H[k]$ are defined simply by sampling $H_d(e^{j\omega})$:
\begin{equation}
H[k] = \left. H_d(e^{j\omega}) \right|_{\omega = \frac{2\pi k}{N}} = H_d\left( e^{j \frac{2\pi k}{N}} \right)
\label{eq:sample_definition}
\end{equation}
The finite-length impulse response $h[n]$ ($0 \le n \le N-1$) is then obtained directly via the $N$-point **Inverse Discrete Fourier Transform (IDFT)**:
\begin{equation}
\mathbf{h[n] = \frac{1}{N} \sum_{k=0}^{N-1} H[k] \, e^{j \frac{2\pi k n}{N}}, \quad 0 \le n \le N-1}
\label{eq:idft_basic}
\end{equation}

\subsection{2.2 The Continuous Interpolation Formula}
Because $h[n]$ is uniquely defined by the IDFT in \eqref{eq:idft_basic}, its continuous $z$-transform $H(z) = \sum_{n=0}^{N-1} h[n] z^{-n}$ can be expressed directly in terms of the frequency samples $H[k]$:
\begin{align}
H(z) &= \sum_{n=0}^{N-1} \left( \frac{1}{N} \sum_{k=0}^{N-1} H[k] \, e^{j \frac{2\pi k n}{N}} \right) z^{-n} \nonumber \\
     &= \frac{1}{N} \sum_{k=0}^{N-1} H[k] \sum_{n=0}^{N-1} \left( e^{j \frac{2\pi k}{N}} z^{-1} \right)^n
\end{align}
Applying the finite geometric series identity $\sum_{n=0}^{N-1} a^n = \frac{1 - a^N}{1 - a}$ with $a = e^{j \frac{2\pi k}{N}} z^{-1}$:
\begin{equation}
a^N = \left( e^{j \frac{2\pi k}{N}} z^{-1} \right)^N = e^{j 2\pi k} z^{-N} = 1 \cdot z^{-N} = z^{-N}
\end{equation}
Substituting this yields the fundamental **Frequency Sampling Interpolation Formula**:
\begin{equation}
\mathbf{H(z) = \frac{1 - z^{-N}}{N} \sum_{k=0}^{N-1} \frac{H[k]}{1 - e^{j \frac{2\pi k}{N}} z^{-1}}}
\label{eq:interpolation_z}
\end{equation}
Evaluating $H(z)$ on the unit circle ($z = e^{j\omega}$) yields the continuous frequency response:
\begin{equation}
\mathbf{H(e^{j\omega}) = \sum_{k=0}^{N-1} H[k] \, \Phi\left( \omega - \frac{2\pi k}{N} \right)}
\label{eq:interpolation_freq}
\end{equation}
where $\Phi(\omega)$ is the complex **Dirichlet interpolation kernel**:
\begin{equation}
\Phi(\omega) = \frac{1}{N} \frac{1 - e^{-j N \omega}}{1 - e^{-j\omega}} = \frac{1}{N} e^{-j \left(\frac{N-1}{2}\right)\omega} \frac{\sin(N\omega/2)}{\sin(\omega/2)}
\label{eq:dirichlet_kernel}
\end{equation}

\noindent\textbf{Key Insights from the Interpolation Formula:}
\begin{enumerate}
    \item \textbf{Exact Pointwise Matching:} At $\omega = \omega_m = \frac{2\pi m}{N}$, the kernel satisfies:
    \begin{equation}
    \Phi\left( \frac{2\pi (m - k)}{N} \right) = \begin{cases} 1, & m = k \\ 0, & m \ne k \end{cases}
    \end{equation}
    Consequently, $H(e^{j 2\pi m / N}) \equiv H[m]$ exactly. There is zero approximation error at the sampling grid points!
    \item \textbf{Inter-Sample Behavior:} Between the grid points, $H(e^{j\omega})$ is formed by the weighted superposition of shifted Dirichlet kernels $\frac{\sin(N(\omega - \omega_k)/2)}{N\sin((\omega - \omega_k)/2)}$. If adjacent samples differ sharply (e.g. from $1$ to $0$), large inter-sample ripples inevitably emerge!
\end{enumerate}

\section{3. LINEAR-PHASE CONSTRAINTS AND REAL IDFT SYNTHESIS}

For the designed FIR filter to possess a strictly linear phase and constant group delay $\tau = (N-1)/2$, the frequency samples $H[k]$ must satisfy specific magnitude and phase symmetry constraints.

\subsection{3.1 Linear-Phase Sample Constraints}
Let the complex frequency samples be written in polar form:
\begin{equation}
H[k] = |H[k]| \, e^{j \angle H[k]} = |H[k]| \, e^{j \theta_k}
\end{equation}
For a real-valued, symmetric impulse response $h[n] = h[N-1-n]$:
\begin{enumerate}
    \item \textbf{Magnitude Symmetry (Even Symmetry):}
    \begin{equation}
    \mathbf{|H[N-k]| = |H[k]|}, \quad k = 1, 2, \dots, N-1
    \label{eq:mag_symmetry}
    \end{equation}
    \item \textbf{Phase Angle Assignment (Linear Phase):}
    The phase must be a strictly linear function of frequency:
    \begin{equation}
    \mathbf{\theta_k = \angle H[k] = -\left( \frac{N-1}{2} \right) \omega_k = -\left( \frac{N-1}{2} \right) \frac{2\pi k}{N}}
    \label{eq:phase_assignment}
    \end{equation}
    Notice that at DC ($k=0$):
    \begin{equation}
    \theta_0 = 0 \implies H[0] \text{ is strictly real and positive.}
    \end{equation}
    \item \textbf{Hermitian Conjugate Symmetry:}
    Combining \eqref{eq:mag_symmetry} and \eqref{eq:phase_assignment} verifies that:
    \begin{equation}
    H[N-k] = H^*[k]
    \label{eq:hermitian_sym}
    \end{equation}
    which guarantees that the synthesized impulse response $h[n]$ is strictly real-valued.
\end{enumerate}

\subsection{3.2 Derivation of Closed-Form Real IDFT Formulas}
Evaluating the general IDFT equation \eqref{eq:idft_basic} directly involves complex exponentials. By pairing complex conjugate terms $k$ and $N-k$, we can derive a purely real trigonometric summation formula.

For any $k \ge 1$, consider the pair of terms in the IDFT:
\begin{align}
T_k &= H[k] e^{j \frac{2\pi k n}{N}} + H[N-k] e^{j \frac{2\pi(N-k)n}{N}} \nonumber \\
    &= |H[k]| e^{j \left( \theta_k + \frac{2\pi k n}{N} \right)} + |H[k]| e^{-j \left( \theta_k + \frac{2\pi k n}{N} \right)} \nonumber \\
    &= 2 |H[k]| \cos\left( \theta_k + \frac{2\pi k n}{N} \right)
\end{align}
Substituting the linear-phase definition $\theta_k = -\left(\frac{N-1}{2}\right)\frac{2\pi k}{N}$:
\begin{equation}
\theta_k + \frac{2\pi k n}{N} = \frac{2\pi k}{N} \left( n - \frac{N-1}{2} \right)
\end{equation}
Thus, the conjugate pair reduces to:
\begin{equation}
T_k = 2 |H[k]| \cos\left[ \frac{2\pi k}{N} \left( n - \frac{N-1}{2} \right) \right]
\label{eq:cos_pair}
\end{equation}

\subsection{3.3 Explicit Formulas for Type 1 (Odd Length \texorpdfstring{$N$}{N}) Filters}
When $N$ is odd, $(N-1)/2$ is an integer. The DFT bins consist of the DC bin ($k=0$) and $(N-1)/2$ conjugate pairs ($k = 1, 2, \dots, (N-1)/2$). Summing all contributions yields the standard \textbf{Type 1 Frequency Sampling Formula}:
\begin{equation}
\mathbf{h[n] = \frac{1}{N} \left\{ H[0] + 2 \sum_{k=1}^{(N-1)/2} |H[k]| \cos\left[ \frac{2\pi k}{N} \left( n - \frac{N-1}{2} \right) \right] \right\}}, \quad 0 \le n \le N-1
\label{eq:type1_formula}
\end{equation}

\subsection{3.4 Explicit Formulas for Type 2 (Even Length \texorpdfstring{$N$}{N}) Filters}
When $N$ is even, $N/2$ is an integer. The DFT bins consist of $k=0$ (DC), $N/2 - 1$ conjugate pairs ($k = 1, \dots, N/2-1$), and a single unpaired bin at the Nyquist frequency $k = N/2$ ($\omega = \pi$).
For $k = N/2$, the phase is $\theta_{N/2} = -\left(\frac{N-1}{2}\right)\pi$. The Nyquist term in the IDFT evaluates to:
\begin{equation}
\frac{1}{N} H[N/2] \, e^{j \pi \left( n - \frac{N-1}{2} \right)}
\end{equation}
However, as proven in Lecture 21, **all Type 2 linear-phase filters possess an obligatory zero at $\omega = \pi$ ($z = -1$)**! Therefore, for any standard lowpass or bandpass design:
\begin{equation}
\mathbf{H[N/2] = 0}
\end{equation}
With $H[N/2] = 0$, the \textbf{Type 2 Frequency Sampling Formula} simplifies to:
\begin{equation}
\mathbf{h[n] = \frac{1}{N} \left\{ H[0] + 2 \sum_{k=1}^{N/2 - 1} |H[k]| \cos\left[ \frac{2\pi k}{N} \left( n - \frac{N-1}{2} \right) \right] \right\}}, \quad 0 \le n \le N-1
\label{eq:type2_formula}
\end{equation}
where $\tau = (N-1)/2$ is a half-integer.

\section{4. TRANSITION SAMPLE OPTIMIZATION (RABINER--GOLD TECHNIQUE)}

\subsection{4.1 The Severe Sidelobe Ringing of Naive Brickwall Sampling}
If a designer samples an ideal brick-wall lowpass filter directly ($H[k] = 1$ in the passband and $H[k] = 0$ in the stopband), a sharp step discontinuity occurs between the last passband sample and the first stopband sample.
As established by Rabiner and Gold (1970), the continuous reconstructed response between grid points exhibits massive Gibbs-like ringing (Figure \ref{fig:interpolation_ripple}):
\begin{itemize}
    \item Passband ripple: $\approx \pm 0.75$ dB.
    \item Peak stopband ripple: $\mathbf{-16 \text{ dB}}$!
\end{itemize}
An attenuation of $-16$ dB is completely intolerable in communication systems, audio codecs, and instrumentation.

\subsection{4.2 Optimization via Unconstrained Transition Samples}
To dramatically eliminate this ringing, we introduce **one, two, or three intermediate samples in the transition band** between the passband and stopband:
\begin{equation}
H[k] \in (0, 1), \quad k \in \text{Transition Band}
\end{equation}
Rather than being fixed to $0$ or $1$, the value $T_1$ (and $T_2, T_3$) is treated as a free optimization parameter and selected via computer search (e.g. minimax linear programming) to minimize the peak sidelobe ripple across the entire stopband.

\section{5. STEP-BY-STEP WORKED NUMERICAL EXAMPLES}

\subsection{Example 24.1: Manual Design of a 7-Point Type 1 Linear-Phase LPF}
\textbf{Problem Statement:} Design a 7-point ($N=7$) linear-phase FIR lowpass filter using the frequency sampling technique. The desired frequency samples are specified as:
\begin{equation}
|H[k]| = \begin{cases} 1, & k = 0, 1, 6 \\ 0, & k = 2, 3, 4, 5 \end{cases}
\end{equation}
Determine the linear-phase delay $\tau$, calculate all seven filter coefficients $h[0]$ through $h[6]$ using the real-valued IDFT formula, and verify coefficient symmetry.

\vspace{2mm}
\noindent\textbf{Solution:}

\noindent\textbf{Step 1: Identify Filter Parameters and Symmetry Properties}
\begin{itemize}
    \item Length $N = 7$ (Odd $\implies$ Type 1 Linear-Phase FIR filter).
    \item Group delay: $\tau = \frac{N-1}{2} = \frac{7-1}{2} = 3$ samples.
    \item Center frequency index: $\frac{N-1}{2} = 3$.
    \item Symmetrical magnitude check: $|H[6]| = |H[7-1]| = |H[1]| = 1$, $|H[5]| = |H[2]| = 0$, $|H[4]| = |H[3]| = 0$. (Even symmetry is strictly satisfied).
\end{itemize}

\noindent\textbf{Step 2: Apply the Real-Valued IDFT Formula for $N$ Odd}
Using \eqref{eq:type1_formula} with $N=7$ and $\tau = 3$:
\begin{equation}
h[n] = \frac{1}{7} \left\{ H[0] + 2 \sum_{k=1}^{3} |H[k]| \cos\left[ \frac{2\pi k}{7} (n - 3) \right] \right\}
\end{equation}
Substituting $|H[0]| = 1$, $|H[1]| = 1$, and $|H[2]| = |H[3]| = 0$:
\begin{equation}
\mathbf{h[n] = \frac{1}{7} \left\{ 1 + 2 \cos\left[ \frac{2\pi}{7} (n - 3) \right] \right\}}, \quad n = 0, 1, \dots, 6
\end{equation}

\noindent\textbf{Step 3: Point-by-Point Numerical Coefficient Calculation}
\begin{itemize}
    \item \textbf{Center tap ($n = 3$):}
    \begin{equation}
    h[3] = \frac{1}{7} [1 + 2 \cos(0)] = \frac{1 + 2(1)}{7} = \frac{3}{7} \approx \mathbf{0.4286}
    \end{equation}
    \item \textbf{Taps adjacent to center ($n = 2$ and $n = 4$):}
    Argument: $\frac{2\pi}{7} (2 - 3) = -\frac{2\pi}{7} = -51.4286^\circ$. Since $\cos(-\theta) = \cos\theta$:
    \begin{equation}
    \cos\left(\frac{2\pi}{7}\right) \approx 0.623490
    \end{equation}
    \begin{equation}
    h[2] = h[4] = \frac{1 + 2(0.623490)}{7} = \frac{1 + 1.24698}{7} = \frac{2.24698}{7} \approx \mathbf{0.3210}
    \end{equation}
    \item \textbf{Taps at distance 2 ($n = 1$ and $n = 5$):}
    Argument: $\frac{2\pi}{7} (1 - 3) = -\frac{4\pi}{7} = -102.8571^\circ$:
    \begin{equation}
    \cos\left(\frac{4\pi}{7}\right) \approx -0.222521
    \end{equation}
    \begin{equation}
    h[1] = h[5] = \frac{1 + 2(-0.222521)}{7} = \frac{1 - 0.445042}{7} = \frac{0.554958}{7} \approx \mathbf{0.0793}
    \end{equation}
    \item \textbf{Boundary taps ($n = 0$ and $n = 6$):}
    Argument: $\frac{2\pi}{7} (0 - 3) = -\frac{6\pi}{7} = -154.2857^\circ$:
    \begin{equation}
    \cos\left(\frac{6\pi}{7}\right) \approx -0.900969
    \end{equation}
    \begin{equation}
    h[0] = h[6] = \frac{1 + 2(-0.900969)}{7} = \frac{1 - 1.801938}{7} = \frac{-0.801938}{7} \approx \mathbf{-0.1146}
    \end{equation}
\end{itemize}

\noindent\textbf{Step 4: Summary of Designed Impulse Response}

\mathbf{h[n] = \{ \underset{\uparrow}{-0.1146}, \; 0.0793, \; 0.3210, \; 0.4286, \; 0.3210, \; 0.0793, \; -0.1146 \}}

Symmetry check: $h[0] = h[6]$, $h[1] = h[5]$, $h[2] = h[4]$. Even symmetry is confirmed.

\subsection{Example 24.2: Design of a 15-Point LPF with 1 Optimized Transition Sample}
\textbf{Problem Statement:} Design a 15-point ($N=15$) linear-phase FIR lowpass filter using frequency sampling. To achieve high stopband rejection ($A_s \ge 45$ dB), introduce one transition sample $T_1 = 0.38$ at the band edge. The frequency samples are specified as:
\begin{equation}
|H[k]| = \begin{cases} 
1.0, & k = 0, 1, 2, 13, 14 \\ 
0.38, & k = 3, 12 \\ 
0.0, & k = 4, 5, 6, 7, 8, 9, 10, 11 
\end{cases}
\end{equation}
Calculate the filter delay $\tau$, formulate the explicit expression for $h[n]$, and compute all 15 impulse response coefficients.

\vspace{2mm}
\noindent\textbf{Solution:}

\noindent\textbf{Step 1: Filter Classification and Symmetries}
\begin{itemize}
    \item Length $N = 15$ (Odd $\implies$ Type 1 linear-phase filter).
    \item Linear-phase delay: $\tau = \frac{15-1}{2} = 7$ samples.
    \item Center index: $n = 7$.
    \item Maximum index in summation: $\frac{N-1}{2} = 7$.
\end{itemize}

\noindent\textbf{Step 2: Formulate Real IDFT Summation}
Substituting $N=15$, $\tau=7$, and the non-zero magnitude samples into \eqref{eq:type1_formula}:
\begin{align}
h[n] = \frac{1}{15} \bigg\{ 1.0 &+ 2(1.0) \cos\left[\frac{2\pi (1)}{15}(n - 7)\right] + 2(1.0) \cos\left[\frac{2\pi (2)}{15}(n - 7)\right] \nonumber \\
&+ 2(0.38) \cos\left[\frac{2\pi (3)}{15}(n - 7)\right] \bigg\}
\end{align}
Simplifying the numerical coefficients:
\begin{equation}
\mathbf{h[n] = \frac{1}{15} \left\{ 1 + 2\cos\left[\frac{2\pi}{15}(n - 7)\right] + 2\cos\left[\frac{4\pi}{15}(n - 7)\right] + 0.76\cos\left[\frac{2\pi}{5}(n - 7)\right] \right\}}
\end{equation}

\noindent\textbf{Step 3: Pointwise Evaluation Exploiting Symmetry ($h[n] = h[14-n]$)}
\begin{itemize}
    \item \textbf{Center Tap ($n = 7$):}
    \begin{equation}
    h[7] = \frac{1 + 2(1) + 2(1) + 0.76(1)}{15} = \frac{5.76}{15} = \mathbf{0.3840}
    \end{equation}
    \item \textbf{Taps $n = 6$ and $n = 8$ ($n - 7 = \pm 1$):}
    \begin{align}
    \cos(2\pi/15) &= \cos(24^\circ) \approx 0.9135 \nonumber \\
    \cos(4\pi/15) &= \cos(48^\circ) \approx 0.6691 \nonumber \\
    \cos(2\pi/5)  &= \cos(72^\circ) \approx 0.3090 \nonumber \\
    h[6] = h[8] &= \frac{1 + 2(0.9135) + 2(0.6691) + 0.76(0.3090)}{15} = \frac{4.4001}{15} \approx \mathbf{0.2933}
    \end{align}

    \end{align}
    \item \textbf{Taps $n = 4$ and $n = 10$ ($n - 7 = \pm 3$):}
    \begin{align}
    \cos(72^\circ) &\approx 0.3090, \quad \cos(144^\circ) \approx -0.8090, \quad \cos(216^\circ) \approx -0.8090 \nonumber \\
    h[4] = h[10] &= \frac{1 + 2(0.3090) + 2(-0.8090) + 0.76(-0.8090)}{15} = \frac{-0.6148}{15} \approx \mathbf{-0.0410}
    \end{align}
    \item \textbf{Taps $n = 3$ and $n = 11$ ($n - 7 = \pm 4$):}
    \begin{equation}
    h[3] = h[11] \approx \mathbf{-0.0620}
    \end{equation}

    \begin{equation}

    \end{equation}

\subsection{Example 24.3: Design of an 8-Point Type 2 Even-Length LPF}
\textbf{Problem Statement:} Design an 8-point ($N=8$) linear-phase FIR lowpass filter using frequency sampling with samples:
\begin{equation}
|H[k]| = \begin{cases} 1, & k = 0, 1, 7 \\ 0, & k = 2, 3, 4, 5, 6 \end{cases}
\end{equation}
Determine the half-sample delay $\tau$, formulate the Type 2 IDFT equation, compute all 8 impulse response coefficients, and verify even symmetry about the half-sample center.

\vspace{2mm}
\noindent\textbf{Solution:}

\noindent\textbf{Step 1: Filter Classification and Nyquist Bin Enforcement}
\begin{itemize}
    \item Length $N = 8$ (Even $\implies$ Type 2 Linear-Phase FIR filter).
    \item Linear-phase delay: $\tau = \frac{N-1}{2} = \frac{8-1}{2} = \mathbf{3.5 \text{ samples}}$ (Half-integer delay).
    \item The Nyquist bin is $k = N/2 = 4$. By Type 2 linear-phase theory, $H[4] = 0$ is mandatory (guaranteeing a transmission zero at $\omega = \pi$).
    \item Summation upper limit: $N/2 - 1 = 8/2 - 1 = 3$.
\end{itemize}

\noindent\textbf{Step 2: Formulate Real IDFT Summation for $N$ Even}
Using \eqref{eq:type2_formula} with $N=8$ and $\tau = 3.5$:
\begin{equation}
h[n] = \frac{1}{8} \left\{ H[0] + 2 \sum_{k=1}^{3} |H[k]| \cos\left[ \frac{2\pi k}{8} (n - 3.5) \right] \right\}
\end{equation}
Since $|H[0]| = 1$, $|H[1]| = 1$, and $|H[2]| = |H[3]| = 0$:
\begin{equation}
\mathbf{h[n] = \frac{1}{8} \left\{ 1 + 2 \cos\left[ \frac{\pi}{4} (n - 3.5) \right] \right\}}, \quad n = 0, 1, \dots, 7
\end{equation}

\noindent\textbf{Step 3: Point-by-Point Numerical Coefficient Calculation}
\begin{itemize}
    \item \textbf{Inner Taps ($n = 3$ and $n = 4$):}
    For $n=3$, $n - 3.5 = -0.5$. Argument: $\frac{\pi}{4}(-0.5) = -\frac{\pi}{8} = -22.5^\circ$:
    \begin{equation}
    \cos\left(\frac{\pi}{8}\right) = \cos(22.5^\circ) = \frac{\sqrt{2 + \sqrt{2}}}{2} \approx 0.923880
    \end{equation}
    \begin{equation}
    h[3] = h[4] = \frac{1 + 2(0.923880)}{8} = \frac{1 + 1.84776}{8} = \frac{2.84776}{8} \approx \mathbf{0.3560}
    \end{equation}
    \item \textbf{Taps at distance 1 ($n = 2$ and $n = 5$):}
    For $n=2$, $n - 3.5 = -1.5$. Argument: $\frac{\pi}{4}(-1.5) = -\frac{3\pi}{8} = -67.5^\circ$:
    \begin{equation}
    \cos\left(\frac{3\pi}{8}\right) = \cos(67.5^\circ) = \frac{\sqrt{2 - \sqrt{2}}}{2} \approx 0.382683
    \end{equation}
    \begin{equation}
    h[2] = h[5] = \frac{1 + 2(0.382683)}{8} = \frac{1 + 0.765367}{8} = \frac{1.765367}{8} \approx \mathbf{0.2207}
    \end{equation}
    \item \textbf{Taps at distance 2 ($n = 1$ and $n = 6$):}
    For $n=1$, $n - 3.5 = -2.5$. Argument: $\frac{\pi}{4}(-2.5) = -\frac{5\pi}{8} = -112.5^\circ$:
    \begin{equation}
    \cos\left(\frac{5\pi}{8}\right) = -\cos(67.5^\circ) \approx -0.382683
    \end{equation}
    \begin{equation}
    h[1] = h[6] = \frac{1 + 2(-0.382683)}{8} = \frac{1 - 0.765367}{8} = \frac{0.234633}{8} \approx \mathbf{0.0293}
    \end{equation}
    \item \textbf{Outer Boundary Taps ($n = 0$ and $n = 7$):}
    For $n=0$, $n - 3.5 = -3.5$. Argument: $\frac{\pi}{4}(-3.5) = -\frac{7\pi}{8} = -157.5^\circ$:
    \begin{equation}
    \cos\left(\frac{7\pi}{8}\right) = -\cos(22.5^\circ) \approx -0.923880
    \end{equation}
    \begin{equation}
    h[0] = h[7] = \frac{1 + 2(-0.923880)}{8} = \frac{1 - 1.84776}{8} = \frac{-0.84776}{8} \approx \mathbf{-0.1060}
    \end{equation}
\end{itemize}

\noindent\textbf{Step 4: Summary and Symmetry Verification}
\begin{equation}
\mathbf{h[n] = \{ \underset{\uparrow}{-0.1060}, \; 0.0293, \; 0.2207, \; 0.3560, \; 0.3560, \; 0.2207, \; 0.0293, \; -0.1060 \}}
\end{equation}
Strict even symmetry about the half-sample center $\tau = 3.5$ is satisfied ($h[n] = h[7-n]$).

\section{6. FACULTY INSIGHTS, COMMON PITFALLS \& EXAM TRAPS}

\begin{itemize}
    \item \textbf{Pitfall 1: Forgetting the Factor of $2$ in the Real IDFT Summation:}
    A common student error in university semester exams is writing $h[n] = \frac{1}{N}[H[0] + \sum |H[k]|\cos(\dots)]$, forgetting the factor of \textbf{$2$} arising from pairing conjugate bins $H[k]$ and $H[N-k]$. This cuts the passband gain in half.
    
    \item \textbf{Pitfall 2: Confusing DFT Index $k$ with Continuous Frequency $\omega$:}
    Remember that sample index $k$ represents frequency $\omega_k = \frac{2\pi k}{N}$ rad/sample. For instance, for $N=16$, $k=2$ corresponds to $\omega = \frac{2\pi(2)}{16} = \frac{\pi}{4} = 0.25\pi$ rad/sample.
    
    \item \textbf{Pitfall 3: Failing to Enforce $H[N/2] = 0$ for Even-Length Type 2 Filters:}
    When $N$ is even, setting $H[N/2] \ne 0$ contradicts the inherent zero at $\omega = \pi$ of Type 2 filters, causing severe distortion and complex coefficients if the phase is not handled with extreme care.
    
    \item \textbf{Pitfall 4: Assuming Zero Ripple Between Sampling Points:}
    Because $H(e^{j\omega_k}) = H[k]$ is exact at grid frequencies, students frequently assume the filter perfectly passes or stops signals between bins. In reality, large interpolation ripples occur between grid points unless transition samples are introduced.
    
    \item \textbf{Pitfall 5: Neglecting the Transition Bandwidth Penalty of Transition Samples:}
    While introducing transition samples ($T_1, T_2$) dramatically improves stopband rejection (e.g., from $-16$ dB to $-45$ dB or $-75$ dB), each transition sample widens the transition region by one frequency bin spacing ($\Delta\omega = 2\pi/N$). To satisfy both a narrow transition band and deep stopband rejection, the total filter length $N$ must be increased proportionally.
\end{itemize}
