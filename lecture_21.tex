\section*{2. MATHEMATICAL FOUNDATIONS}

\subsection*{2.1 The 4 Types of Linear-Phase FIR Filters}

\end{enumerate}
| Type | Symmetry | Length $N$ (Order $M=N-1$) | Amplitude Function $A(\omega)$ | Mandatory Boundary Zeros | Suitable Applications |
| :---: | :---: | :---: | :---: | :---: | :---: |
| \textbf{Type I} | Symmetric ($h[n] = h[N-1-n]$) | $N$ Odd ($M$ Even) | $\sum a[k] \cos(\omega k)$ | None | LPF, HPF, BPF, BSF (Universal) |
| \textbf{Type II} | Symmetric ($h[n] = h[N-1-n]$) | $N$ Even ($M$ Odd) | $\sum b[k] \cos(\omega(k-1/2))$ | Zero at $\omega = \pi$ ($z = -1$) | LPF, BPF only |
| \textbf{Type III} | Antisymmetric ($h[n] = -h[N-1-n]$) | $N$ Odd ($M$ Even) | $\sum c[k] \sin(\omega k)$ | Zeros at $\omega = 0$ ($z=1$) \& $\omega = \pi$ ($z=-1$) | BPF, Differentiator, Hilbert |
| \textbf{Type IV} | Antisymmetric ($h[n] = -h[N-1-n]$) | $N$ Even ($M$ Odd) | $\sum d[k] \sin(\omega(k-1/2))$ | Zero at $\omega = 0$ ($z = 1$) | HPF, BPF, Differentiator, Hilbert |

\subsection*{2.2 Proof of Mandatory Zeros}
\begin{itemize}
\item \textbf{Type II at $\omega = \pi$ ($z = -1$):}
\end{itemize}
  $$ H(-1) = \sum_{n=0}^{N-1} h[n] (-1)^{-n} = \sum_{n=0}^{N/2-1} h[n] [(-1)^{-n} + (-1)^{-(N-1-n)}] $$
  Since $N$ is even, $N-1$ is odd $\implies (-1)^{-(N-1-n)} = -(-1)^{-n} \implies H(-1) = 0$.
  Therefore, a Type II filter always has a zero at $\omega = \pi$ and can \textbf{never} be used as a Highpass or Bandstop filter.

\subsection*{2.3 Zero Quadruplet Locations}
Because $h[n]$ is real and symmetric/antisymmetric:
$$ H(z) = \pm z^{-(N-1)} H(z^{-1}) $$
If $z_0 = r e^{j\theta}$ is a zero, then:
\begin{enumerate}
\item $z_0^* = r e^{-j\theta}$ is a zero (conjugate symmetry for real coefficients).
\item $1/z_0 = \frac{1}{r} e^{-j\theta}$ is a zero (linear phase symmetry).
\item $1/z_0^* = \frac{1}{r} e^{j\theta}$ is a zero.
\end{enumerate}
Zeros off the unit circle always occur in groups of 4 (quadruplets).

\section*{3. WORKED NUMERICAL EXAMPLES}

\subsection*{Example 21.1: FIR Type Identification \& Zero Locations}
\textbf{Problem:} An FIR filter has transfer function $H(z) = 1 + 2z^{-1} + 3z^{-2} + 2z^{-3} + z^{-4}$.
(a) Determine its FIR Type, filter length $N$, and group delay $\tau_g$.
(b) Can this filter be used as a Highpass filter?

\textbf{Solution:}
\textbf{(a)} Coefficients: $h = [1, 2, 3, 2, 1]$.
\begin{itemize}
\item Length $N = 5$ (Odd), Order $M = 4$ (Even).
\item Symmetry: $h[0]=h[4]=1, \; h[1]=h[3]=2, \; h[2]=3 \implies$ \textbf{Symmetric}.
\item Therefore, it is a \textbf{Type I FIR Filter}.
\item Group delay: $\tau_g = \frac{N-1}{2} = \frac{5-1}{2} = 2 \text{ samples}$.

\end{itemize}
\textbf{(b)} Highpass Feasibility:
Check $H(e^{j\pi}) = H(-1) = 1 - 2 + 3 - 2 + 1 = 1 \ne 0$.
Since Type I has no mandatory zeros at $\omega = \pi$, it can theoretically operate as a Highpass filter (though this specific impulse response has $H(e^{j0}) = 9$ and $H(e^{j\pi}) = 1$, making it a Lowpass filter).

import numpy as np

h = np.array([1, 2, 3, 2, 1])
w = np.linspace(0, np.pi, 500)
H = np.polyval(h[::-1], np.exp(-1j * w))

\title{\textbf{EE3621: Digital Signal Processing} \\ \Large Phase should be exactly -2*w \\ \large (III B.Tech EEE)}\maketitle
phase = np.unwrap(np.angle(H))
group_delay = -np.diff(phase) / np.diff(w)
print(f"Mean Group Delay: {np.mean(group_delay):.4f} samples (Theoretical = 2.0)")
