\section*{2. MATHEMATICAL FOUNDATIONS}

\subsection*{2.1 The Matched Z-Transform (MZT)}
Given analog factored transfer function:
$$ H_a(s) = K \frac{\prod_{i=1}^{M} (s - z_i)}{\prod_{k=1}^{N} (s - p_k)} $$
The discrete transfer function is formed by mapping each factor:
$$ H(z) = K_d \frac{\prod_{i=1}^{M} (1 - e^{z_i T_d} z^{-1})}{\prod_{k=1}^{N} (1 - e^{p_k T_d} z^{-1})} (1 + z^{-1})^{N - M} $$
Where $K_d$ is chosen to match the DC gain or passband peak.

\section*{3. WORKED NUMERICAL EXAMPLES}

\subsection*{Example 29.1: Simple Matched Z-Transform Design}
\textbf{Problem:} Design a digital IIR lowpass filter using the Matched Z-Transform method given the analog transfer function:
$$ H_a(s) = \frac{1}{s + 2} $$
Assume a sampling frequency of $F_s = 10 \text{ Hz}$ (or sampling period $T_d = 0.1 \text{ s}$). Set the DC gain of the digital filter to match the DC gain of the analog filter.

\textbf{Solution:}
\begin{enumerate}
\item \textbf{Identify poles and zeros of $H_a(s)$:}
   The analog transfer function has no finite zeros ($M=0$) and one pole at $s_1 = -2$ ($N=1$).
\item \textbf{Map poles and zeros to the z-plane:}
   The single analog pole maps to a digital pole:
   $$ z_1 = e^{s_1 T_d} = e^{-2 \times 0.1} = e^{-0.2} \approx 0.8187 $$
\item \textbf{Formulate the unscaled digital transfer function:}
   According to the MZT formulation, we map finite poles and zeros and add $(1 + z^{-1})^{N-M} = (1 + z^{-1})^1$ to the numerator to map the analog zero at infinity to $z = -1$ (the Nyquist frequency):
   $$ H(z) = K_d \frac{1 + z^{-1}}{1 - e^{-0.2} z^{-1}} = K_d \frac{1 + z^{-1}}{1 - 0.8187 z^{-1}} $$
\item \textbf{Determine the gain factor $K_d$:}
   Match the DC gain ($s=0$ or $\omega=0$) of the analog filter:
   $$ H_a(0) = \frac{1}{0 + 2} = 0.5 $$
   Evaluate the unscaled digital filter at DC ($z=1$):
   $$ H(1) = K_d \frac{1 + 1}{1 - 0.8187} = K_d \frac{2}{0.1813} = 11.0314 K_d $$
   Equating the two:
   $$ 11.0314 K_d = 0.5 \implies K_d \approx 0.0453 $$
\item \textbf{Final Digital Transfer Function:}
   $$ H(z) = \frac{0.0453(1 + z^{-1})}{1 - 0.8187 z^{-1}} $$
\end{enumerate}
