\section{Derivation of the Convolution Sum}
An LTI system is completely characterized by its impulse response $h[n] = \mathcal{T}\{\delta[n]\}$.
By the sifting property, any arbitrary input $x[n]$ is expressed as a weighted sum of impulses:
\begin{equation}
x[n] = \sum_{k=-\infty}^{\infty} x[k] \delta[n-k]
\end{equation}
Applying the system operator $\mathcal{T}$ to both sides:
\begin{equation}
y[n] = \mathcal{T}\{x[n]\} = \mathcal{T}\left\{ \sum_{k=-\infty}^{\infty} x[k] \delta[n-k] \right\}
\end{equation}
By the property of \textbf{Linearity} (superposition and scaling), we can move the operator inside the summation:
\begin{equation}
y[n] = \sum_{k=-\infty}^{\infty} x[k] \mathcal{T}\{\delta[n-k]\}
\end{equation}
By the property of \textbf{Time-Invariance}, if the response to a unit impulse $\delta[n]$ is $h[n]$, the response to a shifted impulse $\delta[n-k]$ is a shifted impulse response $h[n-k]$:
\begin{equation}
\mathcal{T}\{\delta[n-k]\} = h[n-k]
\end{equation}
Substituting this back yields the \textbf{Convolution Sum}:
\begin{equation}
y[n] = \sum_{k=-\infty}^{\infty} x[k] h[n-k] = x[n] * h[n]
\end{equation}
This is the convolution sum, illustrated in Figure~\ref{fig:conv_steps}.

\section{Properties of Discrete-Time Convolution}
\subsection{Commutativity}
\begin{equation}
x[n] * h[n] = h[n] * x[n]
\end{equation}
\textit{Proof:} Let $m = n-k \Rightarrow k = n-m$. As $k \to \infty, m \to -\infty$, and as $k \to -\infty, m \to \infty$:
\begin{equation}
x[n] * h[n] = \sum_{k=-\infty}^{\infty} x[k]h[n-k] = \sum_{m=-\infty}^{\infty} x[n-m]h[m] = \sum_{m=-\infty}^{\infty} h[m]x[n-m] = h[n]*x[n]
\end{equation}

\subsection{Associativity and Distributivity}
\begin{align}
(x[n] * h_1[n]) * h_2[n] &= x[n] * (h_1[n] * h_2[n]) \\
x[n] * (h_1[n] + h_2[n]) &= x[n] * h_1[n] + x[n] * h_2[n]
\end{align}

\subsection{Shifting Property}
If $y[n] = x[n] * h[n]$, then:
\begin{equation}
x[n-n_1] * h[n-n_2] = y[n - n_1 - n_2]
\end{equation}

\section{Causality, BIBO Stability \& Step Response}
\subsection{Causality}
An LTI system is causal if and only if:
\begin{equation}
h[n] = 0 \quad \forall n < 0
\end{equation}
For a causal system and causal input, the convolution simplifies to:
\begin{equation}
y[n] = \sum_{k=0}^{n} x[k] h[n-k]
\end{equation}

\subsection{BIBO Stability}
An LTI system is BIBO stable if and only if its impulse response is absolutely summable:
\begin{equation}
S = \sum_{n=-\infty}^{\infty} |h[n]| < \infty
\end{equation}
\textit{Proof (Sufficiency):} If the input is bounded, i.e., $|x[n]| \le M_x < \infty$, then:
\begin{equation}
|y[n]| = \left| \sum_{k=-\infty}^{\infty} x[n-k] h[k] \right| \le \sum_{k=-\infty}^{\infty} |x[n-k]| |h[k]| \le M_x \sum_{k=-\infty}^{\infty} |h[k]| = M_x S < \infty
\end{equation}
Since $S < \infty$, the output is bounded. Stability profiles are shown in Figure~\ref{fig:lti_stability}.

\subsection{Step Response}
The unit step response is defined as:
\begin{equation}
s[n] = h[n] * u[n] = \sum_{k=-\infty}^{n} h[k]
\end{equation}
The impulse response is recovered by:
\begin{equation}
h[n] = s[n] - s[n-1]
\end{equation}

\section{Constant-Coefficient Difference Equations}
Discrete-time LTI recursive systems are described by difference equations:
\begin{equation}
y[n] + \sum_{k=1}^{N} a_k y[n-k] = \sum_{m=0}^{M} b_m x[n-m]
\end{equation}
The total solution is $y[n] = y_h[n] + y_p[n]$.

\subsection{Homogeneous Solution $y_h[n]$}
Set $x[n] = 0$, yielding the homogeneous difference equation. Assume $y_h[n] = C \lambda^n$:
\begin{equation}
\lambda^N + a_1\lambda^{N-1} + \dots + a_N = 0
\end{equation}
For distinct roots $\lambda_k$, the homogeneous solution is:
\begin{equation}
y_h[n] = \sum_{k=1}^{N} C_k \lambda_k^n
\end{equation}

\subsection{Particular Solution $y_p[n]$}
Represented by trial functions corresponding to the inputs, e.g., $y_p[n] = K_p$ for constant inputs, or $y_p[n] = K_p \beta^n$ for exponential inputs $x[n] = A \beta^n$.

\section{Detailed Worked Numerical Examples}

\subsection{Example 1: Convolution of Finite Sequences}
Find the convolution $y[n] = x[n] * h[n]$ for $x[n] = \{1, \mathbf{2}, 1\}$ and $h[n] = \{\mathbf{1}, 1\}$.
\begin{itemize}
    \item $x[n]$ range: $n \in [-1, 1]$, length $L_x = 3$.
    \item $h[n]$ range: $n \in [0, 1]$, length $L_h = 2$.
    \item Output range: starts at $-1 + 0 = -1$, ends at $1 + 1 = 2$, length $L_y = 3 + 2 - 1 = 4$.
\end{itemize}
Calculating index-by-index:
\begin{align*}
y[-1] &= x[-1]h[0] = 1 \cdot 1 = 1 \\
y[0]  &= x[-1]h[1] + x[0]h[0] = 1 \cdot 1 + 2 \cdot 1 = 3 \\
y[1]  &= x[0]h[1] + x[1]h[0] = 2 \cdot 1 + 1 \cdot 1 = 3 \\
y[2]  &= x[1]h[1] = 1 \cdot 1 = 1
\end{align*}
Result: $y[n] = \{1, \mathbf{3}, 3, 1\}$.

\subsection{Example 2: Convolution of Causal Exponential Sequences}
Compute $y[n] = x[n] * h[n]$ for $x[n] = a^n u[n]$ and $h[n] = b^n u[n]$ ($a \neq b$).
\begin{equation*}
y[n] = \sum_{k=0}^{n} a^k b^{n-k} = b^n \sum_{k=0}^{n} (a/b)^k = b^n \left( \frac{1 - (a/b)^{n+1}}{1 - a/b} \right) = \frac{b^{n+1} - a^{n+1}}{b - a} u[n] = \frac{a^{n+1} - b^{n+1}}{a - b} u[n]
\end{equation*}

\subsection{Example 3: Difference Equation Homogeneous and Particular Solutions}
Solve $y[n] - 0.5 y[n-1] = u[n]$ under initial rest conditions.
\begin{enumerate}
    \item Homogeneous solution: $y_h[n] = C(0.5)^n$.
    \item Particular solution: Try $y_p[n] = K \Rightarrow K - 0.5K = 1 \Rightarrow K = 2$.
    \item Complete solution: $y[n] = C(0.5)^n + 2$ for $n \ge 0$.
    \item Apply initial conditions: under initial rest, $y[-1]=0 \Rightarrow y[0] = 0.5(0) + 1 = 1$.
    \item At $n=0$: $y[0] = C + 2 = 1 \Rightarrow C = -1$.
    \item Result: $y[n] = (2 - 0.5^n)u[n]$.
\end{enumerate}

\subsection{Example 4: Stability Test}
Prove if $h[n] = (0.9)^n u[n]$ is stable.
\begin{equation*}
S = \sum_{n=0}^{\infty} |0.9|^n = \frac{1}{1 - 0.9} = 10 < \infty \quad \Longrightarrow \text{System is stable.}
\end{equation*}

\subsection{Example 5: Step Response calculation}
Calculate the step response $s[n]$ of a system with impulse response $h[n] = \delta[n] - \delta[n-1]$.
\begin{equation*}
s[n] = \sum_{k=-\infty}^{n} (\delta[k] - \delta[k-1]) = u[n] - u[n-1] = \delta[n]
\end{equation*}
