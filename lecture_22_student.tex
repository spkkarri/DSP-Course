\section{Core Concept \& Principle of Windowing}

\subsection{The Truncation Problem}
An ideal brick-wall filter (e.g. ideal lowpass with cutoff $\omega_c$) has a frequency response $H_d(e^{j\omega})$. Its impulse response $h_d[n]$ is infinite in duration (IIR) and non-causal:
\begin{equation}
h_{d,\text{LP}}[n] = \frac{1}{2\pi}\int_{-\omega_c}^{\omega_c} 1 \cdot e^{j\omega n} d\omega = \frac{\sin(\omega_c n)}{\pi n}, \quad -\infty < n < \infty
\end{equation}
To realize a physical causal FIR filter of length $N$, we:
\begin{enumerate}
    \item Shift the center of symmetry to $\tau = \alpha = \frac{N-1}{2}$ to ensure causality.
    \item Multiply $h_d[n]$ pointwise by a finite-duration window sequence $w[n]$ ($0 \le n \le N-1$):
    \begin{equation}
    \mathbf{h[n] = h_d[n] \cdot w[n]}, \quad 0 \le n \le N-1
    \end{equation}
\end{enumerate}

\subsection{Frequency-Domain Convolution \& Gibbs Phenomenon}
By the DTFT modulation property, time-domain multiplication corresponds to periodic convolution in frequency:
\begin{equation}
H(e^{j\omega}) = \frac{1}{2\pi} H_d(e^{j\omega}) \circledast W(e^{j\omega}) = \frac{1}{2\pi} \int_{-\pi}^{\pi} H_d(e^{j\theta}) W(e^{j(\omega - \theta)}) d\theta
\end{equation}
\begin{itemize}
    \item \textbf{Mainlobe Width of $W(e^{j\omega})$:} Spreads out the ideal vertical transition into a gradual transition band $\Delta\omega$.
    \item \textbf{Sidelobes of $W(e^{j\omega})$:} Cause oscillations (ringing) in the passband and stopband.
    \item \textbf{Gibbs Phenomenon:} Abrupt rectangular truncation results in a fixed $8.95\%$ overshoot ($\approx -21$ dB peak sidelobe) near the discontinuity, which \textbf{cannot be reduced by increasing $N$}. Smoother, tapered windows are necessary to suppress sidelobes.
\end{itemize}

\begin{tabular}{lll}
\toprule
\textbf{Filter Type} & \textbf{For $n \ne \tau$} & \textbf{Center Tap $n = \tau$} \\
\midrule

\textbf{Bandpass (BPF)} & $h_d[n] = \frac{\sin(\omega_{c2}(n-\tau)) - \sin(\omega_{c1}(n-\tau))}{\pi(n-\tau)}$ & $h_d[\tau] = \frac{\omega_{c2} - \omega_{c1}}{\pi}$ \\
\textbf{Bandstop (BSF)} & $h_d[n] = \frac{\sin(\omega_{c1}(n-\tau)) - \sin(\omega_{c2}(n-\tau))}{\pi(n-\tau)}$ & $h_d[\tau] = 1 - \frac{\omega_{c2} - \omega_{c1}}{\pi}$ \\
\bottomrule
\end{tabular}

\section{5-Step FIR Filter Window Design Procedure}

\begin{enumerate}
    \item \textbf{Identify Required Attenuation $A_s$:} From stopband specification $A_s = -20\log_{10}(\delta_s)$ dB, select the appropriate window from Table \ref{tab:windows} that meets or exceeds $A_s$.
    \item \textbf{Calculate Filter Length $N$:} From the transition bandwidth $\Delta\omega = |\omega_s - \omega_p|$, calculate $N$ using the window's transition width formula. Round up to the \textbf{next odd integer} to ensure Type 1 linear phase.
    \item \textbf{Determine Cutoff Frequency $\omega_c$:} Average the passband and stopband edge frequencies:
    \begin{equation}
    \omega_c = \frac{\omega_p + \omega_s}{2}
    \end{equation}
    \item \textbf{Compute Ideal Impulse Response $h_d[n]$:} Evaluate $h_d[n]$ for $n = 0, 1, \dots, N-1$ using Table \ref{tab:hd_formulas}.
    \item \textbf{Apply Window:} Compute final filter coefficients: $h[n] = h_d[n] \cdot w[n]$.
\end{enumerate}

\section{Standard Solved Exam Example}

\textbf{Problem:} Design a linear-phase FIR lowpass filter with a cutoff frequency $\omega_c = 0.5\pi$ rad/sample and filter length $N = 7$ using a \textbf{Hamming window}.

\vspace{1mm}
\noindent\textbf{Solution:}

\noindent\textbf{Step 1: Parameters \& Center of Symmetry}
Length $N = 7$ (Odd), center $\tau = \frac{7-1}{2} = 3$. Cutoff $\omega_c = 0.5\pi$.

\noindent\textbf{Step 2: Ideal Impulse Response $h_d[n]$}
\begin{equation}
h_d[n] = \frac{\sin(0.5\pi(n-3))}{\pi(n-3)}, \quad n \ne 3; \qquad h_d[3] = \frac{0.5\pi}{\pi} = 0.5
\end{equation}
Evaluating for $n = 0, 1, 2, 3$:
\begin{itemize}
    \item $n = 3$: $h_d[3] = 0.5000$
    \item $n = 2$: $h_d[2] = \frac{\sin(0.5\pi(-1))}{\pi(-1)} = \frac{-1}{-\pi} = \frac{1}{\pi} \approx \mathbf{0.3183}$
    \item $n = 1$: $h_d[1] = \frac{\sin(0.5\pi(-2))}{\pi(-2)} = \frac{\sin(-\pi)}{-2\pi} = \mathbf{0.0000}$
    \item $n = 0$: $h_d[0] = \frac{\sin(0.5\pi(-3))}{\pi(-3)} = \frac{1}{-3\pi} \approx \mathbf{-0.1061}$
\end{itemize}
By symmetry about $\tau = 3$: $h_d[4] = h_d[2] = 0.3183$, $h_d[5] = h_d[1] = 0.0$, $h_d[6] = h_d[0] = -0.1061$.

\noindent\textbf{Step 3: Hamming Window Weights $w[n]$}
\begin{equation}
w[n] = 0.54 - 0.46\cos\left(\frac{2\pi n}{6}\right) = 0.54 - 0.46\cos\left(\frac{\pi n}{3}\right)
\end{equation}
Evaluating for $n = 0, 1, 2, 3$:
\begin{itemize}
    \item $n = 3$: $w[3] = 0.54 - 0.46\cos(\pi) = 0.54 + 0.46 = \mathbf{1.0000}$
    \item $n = 2$: $w[2] = 0.54 - 0.46\cos(2\pi/3) = 0.54 - 0.46(-0.5) = 0.54 + 0.23 = \mathbf{0.7700}$
    \item $n = 1$: $w[1] = 0.54 - 0.46\cos(\pi/3) = 0.54 - 0.46(0.5) = 0.54 - 0.23 = \mathbf{0.3100}$
    \item $n = 0$: $w[0] = 0.54 - 0.46\cos(0) = 0.54 - 0.46 = \mathbf{0.0800}$
\end{itemize}

\noindent\textbf{Step 4: Final Impulse Response Coefficients $h[n] = h_d[n] \cdot w[n]$}
\begin{itemize}
    \item $h[3] = h_d[3] \cdot w[3] = 0.5000 \times 1.0000 = \mathbf{0.5000}$
    \item $h[2] = h[4] = h_d[2] \cdot w[2] = 0.31831 \times 0.7700 = \mathbf{0.2451}$
    \item $h[1] = h[5] = h_d[1] \cdot w[1] = 0.0000 \times 0.3100 = \mathbf{0.0000}$
    \item $h[0] = h[6] = h_d[0] \cdot w[0] = -0.10610 \times 0.0800 = \mathbf{-0.0085}$
\end{itemize}
The final designed filter is:
\begin{equation}
\mathbf{h[n] = \{-0.0085, \; 0.0000, \; 0.2451, \; \underset{\uparrow}{0.5000}, \; 0.2451, \; 0.0000, \; -0.0085\}}
\end{equation}
Symmetry check: $h[n] = h[6-n]$ is satisfied. The filter is a Type 1 linear-phase lowpass filter.

\section{Essential Exam Checklist \& Traps}
\begin{itemize}
    \item[$\square$] \textbf{Center Tap Formula:} For LPF, $h_d[\tau] = \omega_c/\pi$. Do NOT attempt $\frac{\sin(0)}{0}$ without taking the limit!
    \item[$\square$] \textbf{Highpass Inversion:} For HPF, $h_d[\tau] = 1 - \omega_c/\pi$. All other taps are the negative of the LPF taps.
    \item[$\square$] \textbf{Window Selection Rule:} If the specification requires $A_s = 50$ dB, a Hann window ($A_s = 44$ dB) is insufficient. You \textbf{must select Hamming} ($A_s = 53$ dB).
    \item[$\square$] \textbf{Argument in Window Cosine:} Remember the denominator is $N-1$, not $N$: $\cos\left(\frac{2\pi n}{N-1}\right)$.
\end{itemize}
