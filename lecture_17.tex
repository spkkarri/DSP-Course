\section*{2. MATHEMATICAL FOUNDATIONS}

\subsection*{2.1 General IIR Transfer Function}
\end{enumerate}
$$ H(z) = \frac{\sum_{k=0}^{M} b_k z^{-k}}{1 + \sum_{k=1}^{N} a_k z^{-k}} = \frac{Y(z)}{X(z)} $$
Difference Equation:
$$ y[n] = \sum_{k=0}^{M} b_k x[n-k] - \sum_{k=1}^{N} a_k y[n-k] $$

\subsection*{2.2 Direct Form I Realization}
\begin{itemize}
\item Implements feedforward path ($M$ delays) and feedback path ($N$ delays) separately.
\item \textbf{Total Delays:} $M + N$.
\item \textbf{Total Multipliers:} $M + N + 1$.
\item \textbf{Total Adders:} $M + N$.

\subsection*{2.3 Direct Form II (Canonical Realization)}
\end{itemize}
Let $W(z) = \frac{X(z)}{1 + \sum_{k=1}^N a_k z^{-k}}$, then $Y(z) = \left( \sum_{k=0}^M b_k z^{-k} \right) W(z)$.
In the time domain:
$$ w[n] = x[n] - \sum_{k=1}^{N} a_k w[n-k] $$
$$ y[n] = \sum_{k=0}^{M} b_k w[n-k] $$
\begin{itemize}
\item \textbf{Total Delays:} $\max(M, N)$ (Canonical).
\item \textbf{Total Multipliers:} $M + N + 1$.
\item \textbf{Total Adders:} $M + N$.

\subsection*{2.4 Transposed Direct Form II}
\end{itemize}
Applying flow graph reversal to Direct Form II:
$$ y[n] = b_0 x[n] + v_1[n-1] $$
$$ v_k[n] = b_k x[n] - a_k y[n] + v_{k+1}[n-1], \quad k = 1, 2, \dots, N-1 $$
$$ v_N[n] = b_N x[n] - a_N y[n] $$

\section*{3. WORKED NUMERICAL EXAMPLES}

\subsection*{Example 17.1: Direct Form I and II Realization}
\textbf{Problem:} Realize the second-order IIR filter:
$$ H(z) = \frac{2 + 3 z^{-1} + 4 z^{-2}}{1 - 0.6 z^{-1} + 0.25 z^{-2}} $$
in (a) Direct Form I, (b) Direct Form II.

\textbf{Solution:}
Here $b_0 = 2, b_1 = 3, b_2 = 4$ and $a_1 = -0.6, a_2 = 0.25$.
\textbf{(a) Direct Form I:}
\begin{itemize}
\item Requires $M + N = 2 + 2 = 4$ delay elements.
\item Difference Equation: $y[n] = 2x[n] + 3x[n-1] + 4x[n-2] + 0.6y[n-1] - 0.25y[n-2]$.

\end{itemize}
\textbf{(b) Direct Form II (Canonical):}
\begin{itemize}
\item Requires $\max(2, 2) = 2$ delay elements $w[n-1]$ and $w[n-2]$.
\item State Equation: $w[n] = x[n] + 0.6 w[n-1] - 0.25 w[n-2]$.
\item Output Equation: $y[n] = 2 w[n] + 3 w[n-1] + 4 w[n-2]$.
