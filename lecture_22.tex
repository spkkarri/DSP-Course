\section*{2. MATHEMATICAL FOUNDATIONS}

\subsection*{2.1 Ideal Filter Impulse Responses ($\tau = \frac{N-1}{2}$)}
\begin{itemize}
\item \textbf{Lowpass ($\omega_c$):}
\end{itemize}
\end{enumerate}
  $$ h_{d,\text{LP}}[n] = \frac{\sin(\omega_c(n - \tau))}{\pi(n - \tau)} $$
\begin{itemize}
\item \textbf{Highpass ($\omega_c$):}
\end{itemize}
  $$ h_{d,\text{HP}}[n] = \delta[n - \tau] - \frac{\sin(\omega_c(n - \tau))}{\pi(n - \tau)} $$
\begin{itemize}
\item \textbf{Bandpass ($\omega_{c1}, \omega_{c2}$):}
\end{itemize}
  $$ h_{d,\text{BP}}[n] = \frac{\sin(\omega_{c2}(n - \tau))}{\pi(n - \tau)} - \frac{\sin(\omega_{c1}(n - \tau))}{\pi(n - \tau)} $$

\subsection*{2.2 Window Functions Formulation ($0 \le n \le N-1$)}
\begin{enumerate}
\item \textbf{Rectangular Window:}
\end{enumerate}
   $$ w[n] = 1, \quad \Delta\omega = \frac{4\pi}{N}, \quad \text{Peak Sidelobe: } -13 \text{ dB}, \quad A_s = 21 \text{ dB} $$
\begin{enumerate}
\item \textbf{Bartlett (Triangular) Window:}
\end{enumerate}
   $$ w[n] = 1 - \frac{2|n - \tau|}{N-1}, \quad \Delta\omega = \frac{8\pi}{N}, \quad \text{Peak Sidelobe: } -25 \text{ dB}, \quad A_s = 25 \text{ dB} $$
\begin{enumerate}
\item \textbf{Hann (Hanning) Window:}
\end{enumerate}
   $$ w[n] = 0.5 - 0.5\cos\left( \frac{2\pi n}{N-1} \right), \quad \Delta\omega = \frac{8\pi}{N}, \quad \text{Peak Sidelobe: } -31 \text{ dB}, \quad A_s = 44 \text{ dB} $$

\section*{3. WORKED NUMERICAL EXAMPLES}

\subsection*{Example 22.1: FIR Lowpass Filter Design using Hann Window}
\textbf{Problem:} Design a causal FIR lowpass filter with cutoff $\omega_c = \frac{\pi}{4} \text{ rad/sample}$ and length $N = 7$ using a Hann window.

\textbf{Solution:}
\begin{itemize}
\item Length $N = 7 \implies \tau = \frac{7-1}{2} = 3$.
\item Ideal impulse response:
\end{itemize}
  $$ h_d[n] = \frac{\sin(\frac{\pi}{4}(n - 3))}{\pi(n - 3)} $$
  * $h_d[3] = \frac{\omega_c}{\pi} = \frac{\pi/4}{\pi} = 0.2500$
  * $h_d[2] = h_d[4] = \frac{\sin(-\pi/4)}{-\pi} = \frac{\sqrt{2}/2}{\pi} = \frac{0.7071}{\pi} = 0.2251$
  * $h_d[1] = h_d[5] = \frac{\sin(-2\pi/4)}{-2\pi} = \frac{1}{2\pi} = 0.1592$
  * $h_d[0] = h_d[6] = \frac{\sin(-3\pi/4)}{-3\pi} = \frac{0.7071}{3\pi} = 0.0750$
\begin{itemize}
\item \textbf{Hann Window Values ($w[n] = 0.5 - 0.5\cos(\frac{2\pi n}{6})$):}
\end{itemize}
  * $w[3] = 0.5 - 0.5\cos(\pi) = 1.0000$
  * $w[2] = w[4] = 0.5 - 0.5\cos(2\pi/3) = 0.5 - 0.5(-0.5) = 0.7500$
  * $w[1] = w[5] = 0.5 - 0.5\cos(\pi/3) = 0.5 - 0.5(0.5) = 0.2500$
  * $w[0] = w[6] = 0.5 - 0.5\cos(0) = 0.0000$
\begin{itemize}
\item \textbf{Final FIR Filter Coefficients ($h[n] = h_d[n] \cdot w[n]$):}
\end{itemize}
  * $h[0] = h[6] = 0.0750 \times 0.0000 = 0.0000$
  * $h[1] = h[5] = 0.1592 \times 0.2500 = 0.0398$
  * $h[2] = h[4] = 0.2251 \times 0.7500 = 0.1688$
  * $h[3] = 0.2500 \times 1.0000 = 0.2500$
$$ h[n] = \{ \underset{\uparrow}{0.0000}, 0.0398, 0.1688, 0.2500, 0.1688, 0.0398, 0.0000 \} $$

\textbf{Model Answer \& Step-by-Step Marking Rubric:}

\item \textbf{Part (a):}
\end{itemize}

\item \textbf{Part (b):}

\section*{5. PYTHON VERIFICATION SCRIPT}
\begin{verbatim}
import numpy as np

N = 7
tau = 3
wc = np.pi / 4
n = np.arange(N)

\title{\textbf{EE3621: Digital Signal Processing} \\ \Large Ideal LPF \\ \large (III B.Tech EEE)}\maketitle
hd = np.zeros(N)
hd[n != tau] = np.sin(wc * (n[n != tau] - tau)) / (np.pi * (n[n != tau] - tau))
hd[tau] = wc / np.pi

\title{\textbf{EE3621: Digital Signal Processing} \\ \Large Hann window \\ \large (III B.Tech EEE)}\maketitle
w = 0.5 - 0.5 * np.cos(2 * np.pi * n / (N - 1))
h = hd * w

print("Designed FIR Filter Coefficients h[n]:")
print(np.round(h, 4))
\end{verbatim}
