\section*{2. MATHEMATICAL FOUNDATIONS}

\subsection*{2.1 FIR Transfer Function \& Direct Form}
\end{enumerate}
An FIR filter of length $M$ (order $N = M-1$) is described by:
$$ y[n] = \sum_{k=0}^{M-1} b_k x[n-k] = b_0 x[n] + b_1 x[n-1] + \dots + b_{M-1} x[n - M + 1] $$
$$ H(z) = \sum_{k=0}^{M-1} b_k z^{-k} = b_0 + b_1 z^{-1} + b_2 z^{-2} + \dots + b_{M-1} z^{-(M-1)} $$
\begin{itemize}
\item \textbf{Direct Form Structure:} Consists of a tapped delay line of $M-1$ registers, $M$ parallel multiplier branches feeding a multi-input summation bus.

\subsection*{2.2 Transposed Direct Form (Flow Graph Reversal)}
\end{itemize}
By the Flow Graph Reversal Theorem (Tellegen's Theorem for linear networks):
\begin{enumerate}
\item Reverse the direction of all signal branches.
\item Summing nodes become branching nodes; Branching nodes become summing nodes.
\item Interchange the input $x[n]$ and output $y[n]$.
\end{enumerate}
The state equations for the Transposed Direct Form are:
$$ y[n] = b_0 x[n] + v_1[n-1] $$
$$ v_k[n] = b_k x[n] + v_{k+1}[n-1], \quad k = 1, 2, \dots, M-2 $$
$$ v_{M-1}[n] = b_{M-1} x[n] $$
\begin{itemize}
\item \textbf{Engineering Advantage:} The input $x[n]$ is broadcast simultaneously to all multipliers. Adders are distributed between delay registers, eliminating the long critical-path adder tree of the standard direct form and facilitating high-speed pipelining.

\subsection*{2.3 Cascade Realization}
\end{itemize}
Factoring $H(z)$ into a product of second-order real polynomials:
$$ H(z) = b_0 \prod_{k=1}^{K} (1 + \beta_{1k} z^{-1} + \beta_{2k} z^{-2}), \quad K = \lfloor M/2 \rfloor $$
Each second-order section realizes a pair of complex conjugate zeros $(z_k, z_k^*)$:
$$ 1 + \beta_{1k} z^{-1} + \beta_{2k} z^{-2} = (1 - z_k z^{-1})(1 - z_k^* z^{-1}) $$
Where $\beta_{1k} = -2\text{Re}(z_k)$ and $\beta_{2k} = |z_k|^2$.

\section*{3. WORKED NUMERICAL EXAMPLES}

\subsection*{Example 15.1: Direct and Cascade Realization of a 4th-Order FIR Filter}
\textbf{Problem:} Realize the FIR filter given by transfer function:
$$ H(z) = 1 + 2.5 z^{-1} + 2.5 z^{-2} + 1.25 z^{-3} + 0.25 z^{-4} $$
in (a) Direct Form, (b) Transposed Direct Form, and (c) Cascade Form.

\textbf{Solution:}
\textbf{(a) Direct Form:}
$y[n] = x[n] + 2.5 x[n-1] + 2.5 x[n-2] + 1.25 x[n-3] + 0.25 x[n-4]$.
Requires 4 delay elements $z^{-1}$, 5 multipliers (or 4 non-trivial), and 4 two-input adders.

\textbf{(b) Transposed Direct Form:}
$y[n] = x[n] + v_1[n-1]$
$v_1[n] = 2.5 x[n] + v_2[n-1]$
$v_2[n] = 2.5 x[n] + v_3[n-1]$
$v_3[n] = 1.25 x[n] + v_4[n-1]$
$v_4[n] = 0.25 x[n]$.

\textbf{(c) Cascade Form:}
Factor $H(z)$ into two 2nd-order sections:
$$ H(z) = (1 + 1.5 z^{-1} + 0.5 z^{-2})(1 + 1.0 z^{-1} + 0.5 z^{-2}) $$
\begin{itemize}
\item Section 1: $H_1(z) = 1 + 1.5 z^{-1} + 0.5 z^{-2}$ (zeros at $z = -0.5, -1.0$)
\item Section 2: $H_2(z) = 1 + 1.0 z^{-1} + 0.5 z^{-2}$ (zeros at $z = -0.5 \pm j0.5$)
\end{itemize}
The system is realized as the direct cascade of $H_1(z)$ followed by $H_2(z)$.
