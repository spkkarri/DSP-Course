\section*{2. MATHEMATICAL FOUNDATIONS}

\subsection*{2.1 Moving Average Filter}

$$ y[n] = \frac{1}{M} \sum_{k=0}^{M-1} x[n-k] \implies H(z) = \frac{1 - z^{-M}}{M(1 - z^{-1})} $$
$$ H(e^{j\omega}) = \frac{1}{M} \frac{\sin(\omega M / 2)}{\sin(\omega / 2)} e^{-j\omega(M-1)/2} $$
\begin{itemize}
\item $M-1$ zeros are located on the unit circle at $z = e^{j 2\pi k / M}$ for $k = 1, 2, \dots, M-1$.
\item The pole at $z=1$ exactly cancels the zero at $z=1$.
\end{itemize}
